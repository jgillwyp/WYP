-- ============================================================================

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.handle_new_user();

-- WYP migration 015 — private-testing signup gate
-- CONFIRMED RUN by the owner 2026-08-13.
--
-- Background (see decisions log, 2026-08-13): owner wants to test with a
-- small, known group of people without the app being openly self-serve —
-- "the app testing group will not [be at] risk [of] an unexpected
-- expansion." Two new tables plus one SECURITY DEFINER function anon can
-- call from /login before it ever calls signInWithOtp.
--
-- app_settings — a plain key/value control table, one row per switch.
-- Toggle the gate with a single UPDATE, exactly as the owner asked for:
--   update app_settings set value = false where key = 'signup_gate_enabled';
-- turns it back into an open, self-serve app with no code change or
-- redeploy; setting it back to true re-enables the gate.
--
-- beta_allowlist — one row per email invited to create a brand-new
-- account while the gate is on. Managed the same way: a plain INSERT.
--
-- can_create_account(p_email) — the only thing the client ever calls.
-- Deliberately gates *account creation*, not sign-in in general: if the
-- email already exists in auth.users, it always returns true regardless of
-- the allowlist or the gate setting — a returning user (including the
-- owner's own account) is never affected, matching the owner's own scoping
-- ("This should only apply to brand new signups"). Only a genuinely new
-- email, while the gate is on and not on the allowlist, is refused. Reads
-- auth.users directly, which is safe here specifically because this
-- function is SECURITY DEFINER, owned by the same superuser role that
-- creates every function in this file (CLAUDE.md's Database section: "The
-- SQL editor runs as superuser and bypasses RLS") — anon itself is never
-- granted any access to auth.users, app_settings, or beta_allowlist; it
-- can only ever get a yes/no boolean back through this one function.
-- ============================================================================

begin;

create table if not exists app_settings (
  key text primary key,
  value boolean not null
);
alter table app_settings enable row level security;
-- No policies — deny-all for every client role. Only the SECURITY DEFINER
-- function below (running as the table owner) ever reads this table.

insert into app_settings (key, value)
values ('signup_gate_enabled', true)
on conflict (key) do nothing;

create table if not exists beta_allowlist (
  email text primary key,
  added_at timestamptz not null default now(),
  note text
);
alter table beta_allowlist enable row level security;
-- No policies — same reasoning as app_settings above.

create or replace function public.can_create_account(p_email text)
returns boolean
language plpgsql
security definer
set search_path = public, auth
as $$
declare
  v_gate_enabled boolean;
  v_existing boolean;
  v_allowlisted boolean;
begin
  select coalesce(
    (select value from app_settings where key = 'signup_gate_enabled'),
    true
  ) into v_gate_enabled;

  if not v_gate_enabled then
    return true;
  end if;

  -- Returning user (any email already in auth.users) is never gated,
  -- regardless of the allowlist — the gate is about account *creation*.
  select exists(
    select 1 from auth.users where lower(email) = lower(p_email)
  ) into v_existing;

  if v_existing then
    return true;
  end if;

  select exists(
    select 1 from beta_allowlist where lower(email) = lower(p_email)
  ) into v_allowlisted;

  return v_allowlisted;
end;
$$;

revoke all on function public.can_create_account(text) from public;
grant execute on function public.can_create_account(text) to anon, authenticated;

commit;

-- First attempt failed when the owner ran it (see correction below);
-- corrected version CONFIRMED RUN by the owner 2026-08-13.

  description text,
  due_date date,
  due_time time,
  done_date date,
  created_at timestamptz,
  owner_name text,
  dialog_count bigint
)
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
begin
  if v_email = '' then
    return; -- no email claim on the session; nothing to match
  end if;

  return query
    select
      r.id, r.description, r.due_date, r.due_time, r.done_date, r.created_at,
      p.display_name as owner_name,
      (select count(*) from public.dialog d where d.request_id = r.id) as dialog_count
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    left join public.profiles p on p.id = r.owner_id
    where lower(trim(c.email)) = v_email
    order by r.due_date desc nulls last;
end;
$$;

revoke all on function public.get_received_requests() from public;
grant execute on function public.get_received_requests() to authenticated;

create or replace function public.get_request_by_token(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         ct.display_name as contact_name,
         r.link_expires_at, r.link_revoked_at
    into v_request
    from public.requests r
    left join public.profiles p on p.id = r.owner_id
    left join public.contacts ct on ct.id = r.contact_id
    where r.link_token_hash = v_hash;

  if v_request.id is null
     or v_request.link_revoked_at is not null
     or v_request.link_expires_at < now()
  then
    raise exception 'This link is no longer valid.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', v_request.id, v_request.id, 'viewed', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'contact_name', v_request.contact_name,
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_request_by_token(text) from public;
grant execute on function public.get_request_by_token(text) to anon, authenticated;

-- ============================================================================
-- WYP migration 025 — public.attachments table
-- DRAFTED 2026-08-14 — CONFIRMED RUN by the owner 2026-08-14.
--
-- Backs Week 5 Priority 3 (Real Attachments) -- see docs/WYP_Attachments_Plan.md
-- for the full design and the owner's answers to its seven open questions
-- (decisions log, 2026-08-14).
--
-- One table for both Requests' real uploaded files and ToDos' "Location"
-- references (a file path or URL the owner types in rather than an actual
-- upload) -- `kind` distinguishes them rather than splitting into two
-- tables. request_id covers both, since ToDos are rows in `requests` too
-- (same precedent as `dialog`).
--
-- uploaded_by is NULLABLE, unlike every other owner/author column in this
-- app so far -- an anonymous Request Response visitor (no auth.users row at
-- all) can add a 'file' attachment, the identical shape of problem
-- dialog.author_user_id already solved the same way for the same reason
-- (migration 004: "nothing to attribute to a real app account"). A parallel
-- uploaded_by_label text column carries the display snapshot either way --
-- the signed-in owner's or recipient's own name, or the anonymous visitor's
-- Contact display_name from get_request_by_token -- matching dialog.who's
-- own "snapshot, not a live join" precedent.
--
-- The two shapes never cross, enforced by a check constraint: a 'file' row
-- must have file_name/storage_path/size_bytes/mime_type all set and both
-- reference columns null; a 'reference' row must have both file columns
-- null and at least one of reference_url/reference_note set.
--
-- RLS is deliberately narrow. SELECT is owner-only (every row on a Request/
-- ToDo the signed-in user owns, whatever its kind) -- Create/Detail screens
-- read attachment metadata straight off this policy without an API round
-- trip. INSERT only allows kind = 'reference' -- a 'file' row can only ever
-- be created by the new app/api/attachments/upload route, which uses the
-- service_role key server-side (never the browser) specifically because
-- Storage uploads for an ANONYMOUS Request Response visitor have no session
-- for RLS to scope to at all -- the same category of problem `dialog`/the
-- response-link functions already solved with SECURITY DEFINER functions,
-- solved here with a server-only API route instead because Storage's own
-- upload API isn't reachable from a SQL function. DELETE matches the
-- owner's resolved answer (decisions log, 2026-08-14): the Request/ToDo
-- owner can always delete any attachment on their own item; a non-owner
-- uploader (a Request's recipient) can only delete their own -- and, since
-- uploaded_by is null for an anonymous upload, an anonymous visitor's own
-- attachment can only ever be removed by the owner, never by themselves
-- (matches this build's scoping: no delete UI is offered on the anonymous
-- Request Response screen at all, see app/api/attachments/delete/route.ts's
-- own header comment). See CLAUDE.md's Database section: "assume the
-- control was bypassed" -- the policy is what actually enforces this, the
-- UI's own delete-button visibility is a courtesy on top.
--
-- No UPDATE policy -- an attachment is added or removed, never edited in
-- place, same shape as `dialog`'s own append-only convention.
--
-- deleted_at exists but is NOT used by the app's own Delete button, which
-- hard-deletes (removes the Storage object and the row) -- see
-- app/api/attachments/delete/route.ts. It's reserved for the future lapse-
-- and-auto-delete job (PRD §6.3, deferred -- open question #5 in the plan
-- doc), which needs to distinguish "this Request once had an attachment,
-- now reclaimed by a tier lapse" from "this attachment was simply removed
-- by the user" -- a hard delete already achieves the second; the tombstone
-- is only meaningful for the first, not built yet.
-- ============================================================================

begin;

create table if not exists public.attachments (
  id                 uuid        primary key default gen_random_uuid(),
  request_id         uuid        not null references public.requests(id) on delete cascade,
  uploaded_by        uuid        references auth.users(id) on delete set null,
  uploaded_by_label  text        not null,
  kind               text        not null default 'file'
                                 check (kind in ('file', 'reference')),
  file_name          text,
  storage_path       text,
  size_bytes         bigint,
  mime_type          text,
  reference_url      text,
  reference_note     text,
  created_at         timestamptz not null default now(),
  deleted_at         timestamptz,   -- reserved for the future lapse-and-auto-delete
                                     -- job; the app's own Delete button hard-deletes
                                     -- instead -- see header comment above
  constraint attachments_shape_check check (
    (kind = 'file'
       and file_name is not null and storage_path is not null
       and size_bytes is not null and mime_type is not null
       and reference_url is null and reference_note is null)
    or
    (kind = 'reference'
       and file_name is null and storage_path is null
       and size_bytes is null and mime_type is null
       and (reference_url is not null or reference_note is not null))
  )
);

comment on column public.attachments.kind is
  '''file'' = a real uploaded file on a Request (Supabase Storage, private bucket). ''reference'' = a ToDo ''Location'' -- a typed path or URL, never stored as bytes. A ''file'' row can only be created by app/api/attachments/upload (service_role, server-only); the RLS insert policy below refuses any client-inserted ''file'' row.';
comment on column public.attachments.uploaded_by is
  'Nullable -- an anonymous Request Response visitor has no auth.users row. See uploaded_by_label for the always-present display snapshot.';
comment on column public.attachments.uploaded_by_label is
  'Display-name snapshot at add time, not a live join -- same precedent as dialog.who (migration 004). Always set, even when uploaded_by is null.';
comment on column public.attachments.storage_path is
  'Path within the private ''attachments'' Storage bucket (migration 026), e.g. {request_id}/{attachment_id}-{file_name}. Null for kind = ''reference''.';
comment on column public.attachments.deleted_at is
  'Reserved for the future lapse-and-auto-delete job -- NOT set by the app''s own Delete button, which hard-deletes the row and the Storage object. See header comment.';

create index if not exists attachments_request_idx on public.attachments (request_id, created_at desc);

alter table public.attachments enable row level security;

create policy "attachments: owners select own"
  on public.attachments for select to authenticated
  using (request_id in (select id from public.requests where owner_id = auth.uid()));

create policy "attachments: owners insert reference rows only"
  on public.attachments for insert to authenticated
  with check (
    kind = 'reference'
    and request_id in (select id from public.requests where owner_id = auth.uid())
    and uploaded_by = auth.uid()
  );

create policy "attachments: owner or own-uploads delete"
  on public.attachments for delete to authenticated
  using (
    uploaded_by = auth.uid()
    or request_id in (select id from public.requests where owner_id = auth.uid())
  );

-- No update policy -- see header comment.

commit;

create function public.get_received_requests()
returns table (
  id uuid,
  description text,
  due_date date,
  due_time time,
  done_date date,
  created_at timestamptz,
  owner_name text,
  owner_request_time_enabled boolean,
  dialog_count bigint,
  attachment_count bigint
)
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
begin
  if v_email = '' then
    return; -- no email claim on the session; nothing to match
  end if;

  return query
    select
      r.id, r.description, r.due_date, r.due_time, r.done_date, r.created_at,
      p.display_name as owner_name,
      coalesce(p.request_time_enabled, true) as owner_request_time_enabled,
      (select count(*) from public.dialog d where d.request_id = r.id) as dialog_count,
      (select count(*) from public.attachments a where a.request_id = r.id and a.kind = 'file') as attachment_count
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    left join public.profiles p on p.id = r.owner_id
    where lower(trim(c.email)) = v_email
    order by r.due_date desc nulls last;
end;
$$;

revoke all on function public.get_received_requests() from public;
grant execute on function public.get_received_requests() to authenticated;

-- DRAFTED 2026-08-14 — CONFIRMED RUN by the owner 2026-08-14.

-- received_archived_at has no such path: RLS on `requests` is owner-only, and
-- the recipient isn't the owner. archive_received_request() is a fourth

-- set_response_done_as_recipient/add_dialog_as_recipient (migration 012) —

-- not tackle an 'Un-Archive' feature - that can be done later") — no reverse

--

-- The owner's own drafted §9.5 replacement text (decisions log, 2026-08-14)
-- says archived records are "no longer displayed" on the Main Screen "while
-- remaining available through Search," so Sent/ToDos/Received are still
-- fetched in full; MainScreen.tsx does the show/hide split client-side based
-- on whether a search query is active, not by ever excluding archived rows
-- from a query. get_received_requests() therefore adds received_archived_at
-- to its own row shape rather than filtering by it — RETURNS TABLE, so
-- drop-then-create (migrations 017/021/027's own lesson, applied here
-- proactively rather than rediscovered a fourth time).
--
-- get_received_request() (singular, the Response Detail RPC) is NOT touched
-- this batch — nothing reads received_archived_at there yet, and adding an
-- unused field would be speculative. Revisit if a future Un-Archive-from-
-- Detail affordance needs it.

begin;

alter table public.requests
  add column if not exists archived_at timestamptz,
  add column if not exists received_archived_at timestamptz;

comment on column public.requests.archived_at is
  'Set when the row''s own owner archives it from their Sent list (a Request) or their only list (a ToDo, contact_id null). Null = not archived. Plain-RLS-writable via the existing owner-update policy, no function needed. Still returned by every existing SELECT — MainScreen.tsx hides an archived row from the resting list client-side but still matches it when a search query is active (PRD §9.5: archived stays findable through Search). Un-Archive (setting back to null) is deferred.';
comment on column public.requests.received_archived_at is
  'Set when the recipient archives this Request from their own Received list — independent of archived_at above, since the sender and recipient are two different accounts each viewing the same row (2026-08-14 per-viewer decision, not a shared flag). Never touched by the owner-side "requests: owners update own" policy; only archive_received_request() (this migration) can set it. Always null for a ToDo (contact_id is null, so there is no recipient to archive it from).';

create index if not exists requests_archived_idx on public.requests (owner_id, archived_at);
create index if not exists requests_received_archived_idx on public.requests (contact_id, received_archived_at);

-- ----------------------------------------------------------------------------
-- archive_received_request — the signed-in recipient's Archive action on
-- their own Received copy of a Request. Same email-match-through-contacts
-- guard as set_response_done_as_recipient/add_dialog_as_recipient (migration
-- 012); only ever sets received_archived_at, touches nothing else on the row.
-- ----------------------------------------------------------------------------
create or replace function public.archive_received_request(p_request_id uuid)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request_id uuid;
begin
  select r.id into v_request_id
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = p_request_id
      and v_email <> ''
      and lower(trim(c.email)) = v_email;

  if v_request_id is null then
    raise exception 'Request not found.';
  end if;

  update public.requests
    set received_archived_at = now()
    where id = v_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values (
      'request', v_request_id, v_request_id, 'archived_by_recipient',
      jsonb_build_object('via', 'recipient_session')
    );
end;
$$;

revoke all on function public.archive_received_request(uuid) from public;
grant execute on function public.archive_received_request(uuid) to authenticated;
