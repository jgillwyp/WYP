-- ============================================================================
-- Migration 081 (2026-10-19) -- DRAFTED, NOT YET CONFIRMED RUN.
-- First-party URL shortener for Request Texting SMS links. Jim, after
-- confirming SMS sending works end to end: "The URL shortening would be
-- helpful" -- the real /r/[token] response link carries a 64-character hex
-- token (https://www.wouldyouplease.com/r/<64 chars>, ~108 characters with
-- the domain), which by itself already exceeds a single 160-character SMS
-- segment before any message text is added, so every text sends as at
-- least two billed segments.
--
-- Deliberately first-party (public.short_links + a redirect under this
-- app's own domain, app/s/[code]/route.ts), not a third-party shortener
-- (bit.ly, tinyurl, etc.): the approved A2P 10DLC campaign's own
-- "Embedded links sample" field was reviewed and approved showing a real
-- wouldyouplease.com/r/... URL -- introducing a generic third-party
-- shortener domain afterward would both diverge from what carriers
-- actually vetted and is a well-known trigger for carrier spam filtering
-- on its own (an unrecognized redirect domain reads as more suspicious
-- than a link under the sender's own, already-trusted domain).
--
-- Security trade-off, flagged rather than silently accepted: a /r/[token]
-- link's 64 hex characters (256 bits) are intentionally that long because
-- CLAUDE.md's own Database section treats it as a durable, unrevoked-by-
-- default capability -- shortening it to a 12-character hex code (48
-- bits) trades a large chunk of that margin for SMS usability. 48 bits is
-- still well into the range generally considered acceptable for a
-- brute-force-resistant short code (stronger than most commercial URL
-- shorteners' own 6-7 character codes), and every short_links row only
-- ever points at an already-sanctioned wouldyouplease.com URL -- enforced
-- server-side in create_short_link() below, not left to whatever the
-- caller supplies, which is what actually closes off an open-redirect
-- risk regardless of code length. short_links rows are never deleted or
-- expired by this migration -- same "don't drop structure, revisit later
-- if it matters" posture this file's own history already takes elsewhere
-- (e.g. last_overdue_nudge_at) -- a cleanup job for old rows is a
-- reasonable future addition, not built here.
--
-- public.short_links is openly readable (anon + authenticated SELECT) --
-- the mapping itself isn't sensitive, only reachable by someone who
-- already has the short link, same exposure a real URL shortener service
-- would have anyway -- but has NO insert/update/delete policy for either
-- role. The only way to create a row is create_short_link(), a SECURITY
-- DEFINER function granted to authenticated (the owner-session routes:
-- send-request/route.ts, send-reminder/route.ts) and service_role (the
-- cron route, which has no user session to run as) -- same two-grant
-- pairing this file's history already uses for issue_request_link /
-- cron_issue_request_link.
-- ============================================================================

create table public.short_links (
  code text primary key,
  target_url text not null,
  created_at timestamptz not null default now()
);

alter table public.short_links enable row level security;

create policy "short_links: anyone can read"
  on public.short_links for select
  using (true);

grant select on public.short_links to anon, authenticated;
grant select, insert on public.short_links to service_role;

create or replace function public.create_short_link(p_target_url text)
returns text
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_code text;
  v_prefix constant text := 'https://www.wouldyouplease.com/';
  v_attempts int := 0;
begin
  if p_target_url is null or left(p_target_url, length(v_prefix)) <> v_prefix then
    raise exception 'create_short_link: target_url must be a wouldyouplease.com URL';
  end if;

  loop
    v_attempts := v_attempts + 1;
    if v_attempts > 10 then
      raise exception 'create_short_link: could not generate a unique code';
    end if;
    v_code := encode(gen_random_bytes(6), 'hex');
    begin
      insert into public.short_links (code, target_url) values (v_code, p_target_url);
      exit;
    exception when unique_violation then
      -- extremely unlikely at 48 bits of entropy -- retry with a fresh code
      null;
    end;
  end loop;

  return v_code;
end;
$$;

revoke all on function public.create_short_link(text) from public;
grant execute on function public.create_short_link(text) to authenticated, service_role;

create function public.request_sms_consent(p_contact_id uuid, p_notify_owner boolean default false)
returns text
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_owner uuid;
  v_token text;
begin
  select owner_id into v_owner from public.contacts where id = p_contact_id;

  if v_owner is null or v_owner <> auth.uid() then
    raise exception 'Not found.';
  end if;

  v_token := encode(gen_random_bytes(32), 'hex');

  update public.contacts
    set sms_consent_token_hash = digest(v_token, 'sha256'),
        sms_consent_requested_at = now(),
        sms_consent_notify_owner = p_notify_owner,
        sms_consent_notified_at = null
    where id = p_contact_id;

  return v_token;
end;
$$;

revoke all on function public.request_sms_consent(uuid, boolean) from public;
grant execute on function public.request_sms_consent(uuid, boolean) to authenticated;

create or replace function public.mark_sms_consent_notified(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_row record;
begin
  update public.contacts c
    set sms_consent_notified_at = now()
    where c.sms_consent_token_hash = v_hash
      and c.sms_consent_notify_owner = true
      and c.sms_consent_given_at is not null
      and c.sms_consent_notified_at is null
    returning c.id, c.owner_id, c.display_name into v_row;

  if v_row.owner_id is null then
    return jsonb_build_object('should_notify', false);
  end if;

  return jsonb_build_object(
    'should_notify', true,
    'owner_id', v_row.owner_id,
    'contact_id', v_row.id,
    'contact_name', v_row.display_name
  );
end;
$$;

revoke all on function public.mark_sms_consent_notified(text) from public;
grant execute on function public.mark_sms_consent_notified(text) to anon, authenticated;

-- ============================================================================
-- Migration 083 (2026-10-20) -- DRAFTED, NOT YET CONFIRMED RUN.
-- Jim, after reviewing the carrier-consent model: approval is tied to the
-- specific phone number, not the Contact more broadly. Editing an already-
-- approved Contact's phone number on Contact Detail needs a fresh approval
-- for the new number, worded differently from the original ask (the old
-- wording presumes first-time consent). Two real bugs would exist without
-- this migration, beyond the missing UI dialog (built separately in
-- ContactDetailForm.tsx this same batch):
--
-- 1. request_sms_consent() never reset sms_consent_given_at. Re-calling it
--    for an already-approved Contact (the new re-request path) would have
--    left sms_consent_given_at still set to its old, now-stale value --
--    contactSmsNumber()'s own eligibility check (app/src/lib/sms.ts) only
--    looks at whether sms_consent_given_at is non-null, so texts would have
--    kept silently going to the OLD approved number for as long as the new
--    approval sat pending, while the UI claimed "awaiting approval." Fixed
--    by setting sms_consent_given_at = null on every call -- a no-op for
--    the existing first-time-request case (already null there), correct
--    for the new re-request case.
--
-- 2. get_contact_for_sms_consent()'s own phone pre-fill coalesced
--    sms_consent_phone (the OLD approved number) before contacts.phone (the
--    number the owner just typed and is actually asking to be approved).
--    For a re-request this showed the wrong number on the opt-in page --
--    flipped to prefer the current phone column, falling back to
--    sms_consent_phone only if that's somehow null. Harmless for the
--    existing first-time-request case: set_sms_consent already keeps both
--    columns in sync once approval completes, so they already match
--    whenever this function is reached under the old logic's assumptions.
--
-- Same arg list and return type on both functions -- plain create or
-- replace applies to each, no drop needed.
-- ============================================================================

create or replace function public.request_sms_consent(p_contact_id uuid, p_notify_owner boolean default false)
returns text
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_owner uuid;
  v_token text;
begin
  select owner_id into v_owner from public.contacts where id = p_contact_id;

  if v_owner is null or v_owner <> auth.uid() then
    raise exception 'Not found.';
  end if;

  v_token := encode(gen_random_bytes(32), 'hex');

  update public.contacts
    set sms_consent_token_hash = digest(v_token, 'sha256'),
        sms_consent_requested_at = now(),
        sms_consent_notify_owner = p_notify_owner,
        sms_consent_notified_at = null,
        -- New (migration 083): clears a stale prior approval so a
        -- re-request (phone number changed) actually goes back to
        -- "awaiting" -- texts must stop going to the old number the
        -- instant a new number is being asked about. No-op for the
        -- existing first-time-request case, already null there.
        sms_consent_given_at = null
    where id = p_contact_id;

  return v_token;
end;
$$;

create or replace function public.get_contact_for_sms_consent(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_contact record;
begin
  select c.phone, c.sms_consent_given_at, c.sms_consent_phone,
         c.sms_opted_out_at, p.display_name as owner_name
    into v_contact
    from public.contacts c
    join public.profiles p on p.id = c.owner_id
    where c.sms_consent_token_hash = v_hash;

  if not found then
    raise exception 'This link is no longer valid.';
  end if;

  return jsonb_build_object(
    -- New (migration 083): prefer the CURRENT phone column -- the number
    -- actually being asked about, which may be newer than whatever was
    -- last approved -- falling back to sms_consent_phone only if phone is
    -- somehow null. Was coalesce(sms_consent_phone, phone), the wrong
    -- order for a re-request after a phone number change.
    'phone', coalesce(v_contact.phone, v_contact.sms_consent_phone),
    'owner_name', v_contact.owner_name,
    'consent_given_at', v_contact.sms_consent_given_at,
    'opted_out_at', v_contact.sms_opted_out_at
  );
end;
$$;

-- ============================================================================
-- Migration 084 (2026-10-21) -- sms_consent_links: a Contact can now hold
-- multiple simultaneously-valid text-messaging-consent tokens, instead of
-- one. DRAFTED, NOT YET CONFIRMED RUN.
--
-- Jim, building the new Resend Approval Request button (ContactDetailForm.tsx,
-- same batch): "I am worried about invalidating the prior approval request.
-- Could there be more than one valid request?" -- exactly the right question,
-- and the answer was yes: request_sms_consent (migration 079) has always
-- overwritten the single contacts.sms_consent_token_hash column, silently
-- killing whatever link was in every earlier-sent email the moment a later
-- one needed a fresh token -- including a Resend, and including the
-- phone-number-change re-request this same screen already does (migration
-- 083). This is the identical bug request_links (migration 075) already
-- fixed once for Request response links -- see that migration's own header
-- comment, copied here almost verbatim with "Request"/"request_links"
-- swapped for "Contact"/"sms_consent_links". That migration's own backfill-
-- then-deprecate approach is mirrored exactly: nothing already working
-- breaks the moment this runs, and the old column is marked deprecated
-- rather than dropped in the same migration that changes behavior.
--
-- Unlike request_links, no expires_at/revoked_at columns -- the original
-- design (migration 079's own header comment) already called this token
-- "durable, reusable" with no expiry concept, and a second, later-approved
-- token resolving to the same already-approved Contact causes no
-- incorrectness (get_contact_for_sms_consent and set_sms_consent both key
-- off the Contact's own shared sms_consent_given_at/sms_opted_out_at
-- fields, not any per-token state) -- so there's nothing yet to revoke.
-- Easy to add later if a real need for it shows up, same as request_links'
-- own still-uncalled revoke_request_link.
-- ============================================================================

create table public.sms_consent_links (
  id uuid primary key default gen_random_uuid(),
  contact_id uuid not null references public.contacts(id) on delete cascade,
  token_hash bytea not null,
  created_at timestamptz not null default now()
);

create index sms_consent_links_contact_id_idx on public.sms_consent_links (contact_id);
create unique index sms_consent_links_token_hash_idx on public.sms_consent_links (token_hash);

alter table public.sms_consent_links enable row level security;
-- No policies, and no anon/authenticated grant -- reachable only through
-- the SECURITY DEFINER functions below, same reasoning request_links'
-- own header comment already gives for that table.

comment on table public.sms_consent_links is
  'One row per text-messaging-consent token ever issued for a Contact (migration 084) -- replaces the single-slot contacts.sms_consent_token_hash (migration 079), which silently invalidated every earlier-sent email''s link each time a later one (a Resend, or a phone-number-change re-request) needed a fresh token. Multiple rows may be simultaneously valid for the same contact_id; get_contact_for_sms_consent/set_sms_consent/set_sms_opt_out/mark_sms_consent_notified now match against any of them, not a single column.';

-- Backfill: preserve every currently-live token so nothing already working
-- breaks the moment this migration runs.
insert into public.sms_consent_links (contact_id, token_hash, created_at)
  select id, sms_consent_token_hash, coalesce(sms_consent_requested_at, now())
  from public.contacts
  where sms_consent_token_hash is not null;

comment on column public.contacts.sms_consent_token_hash is
  'DEPRECATED (migration 084) -- superseded by sms_consent_links, which supports multiple simultaneously-valid tokens per Contact. No longer read or written by any function; left in place, unused, rather than dropped in the same migration that changes behavior.';

-- ----------------------------------------------------------------------------
-- request_sms_consent -- mints into sms_consent_links instead of overwriting
-- the single contacts column. Signature and every other field it touches
-- unchanged (including migration 083's sms_consent_given_at = null reset),
-- so create-or-replace is safe -- every existing caller (AddContactForm.tsx,
-- ContactDetailForm.tsx, including its new Resend button and its
-- phone-change re-request) needs no change.
-- ----------------------------------------------------------------------------
create or replace function public.request_sms_consent(p_contact_id uuid, p_notify_owner boolean default false)
returns text
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_owner uuid;
  v_token text;
begin
  select owner_id into v_owner from public.contacts where id = p_contact_id;

  if v_owner is null or v_owner <> auth.uid() then
    raise exception 'Not found.';
  end if;

  v_token := encode(gen_random_bytes(32), 'hex');

  insert into public.sms_consent_links (contact_id, token_hash)
    values (p_contact_id, digest(v_token, 'sha256'));

  update public.contacts
    set sms_consent_requested_at = now(),
        sms_consent_notify_owner = p_notify_owner,
        sms_consent_notified_at = null,
        sms_consent_given_at = null
    where id = p_contact_id;

  return v_token;
end;
$$;

-- ----------------------------------------------------------------------------
-- get_contact_for_sms_consent -- looks up through sms_consent_links now.

-- returned fields) is otherwise unchanged. Signature unchanged, no drop

create or replace function public.get_contact_for_sms_consent(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_contact record;
begin
  select c.phone, c.sms_consent_given_at, c.sms_consent_phone,
         c.sms_opted_out_at, p.display_name as owner_name
    into v_contact
    from public.sms_consent_links scl
    join public.contacts c on c.id = scl.contact_id
    join public.profiles p on p.id = c.owner_id
    where scl.token_hash = v_hash;

  if not found then
    raise exception 'This link is no longer valid.';
  end if;

  return jsonb_build_object(
    'phone', coalesce(v_contact.phone, v_contact.sms_consent_phone),
    'owner_name', v_contact.owner_name,
    'consent_given_at', v_contact.sms_consent_given_at,
    'opted_out_at', v_contact.sms_opted_out_at
  );
end;
$$;

-- set_sms_consent -- looks up through sms_consent_links now. p_phone must
-- already be E.164-normalized by the caller (TextMessagingForm.tsx), same

-- ----------------------------------------------------------------------------
create or replace function public.set_sms_consent(p_token text, p_phone text)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_contact_id uuid;
begin
  select contact_id into v_contact_id from public.sms_consent_links
    where token_hash = digest(p_token, 'sha256');

  if v_contact_id is null then
    raise exception 'This link is no longer valid.';
  end if;

  update public.contacts
    set sms_consent_given_at = now(),
        sms_consent_phone = p_phone,
        phone = p_phone,
        sms_opted_out_at = null
    where id = v_contact_id;
end;
$$;

-- Migration 085 (2026-10-31) -- issue_request_link_as_recipient(): lets a
-- signed-in Recipient mint their own durable /r/[token] response link, so
-- Response Detail's own "Add to Calendar" can embed a link that works from
-- ANY browser, not just one already signed into the matching account.
-- DRAFTED, NOT YET CONFIRMED RUN.
--
-- Owner-reported, reproducing a limitation flagged (not fixed) in an
-- earlier session: Response Detail's Add to Calendar embedded
-- window.location.href -- the authenticated /requests/[id]/respond page --
-- since that was "the closest equivalent available" at the time, Response
-- Detail having no token of its own the way the mailed /r/[token] link
-- does. Tapping that link from the saved calendar event hands off to a
-- browser; unless that browser's signed-in session happens to be the exact
-- matching Recipient account, get_received_request can't match it and
-- raises "Request not found." -- reproduced live, 2026-10-31.
--
-- Same shape as issue_request_link (migration 008, mints for the owner,
-- gated on auth.uid() = owner_id) and cron_issue_request_link (migration
-- 075, mints for the cron job, no caller to gate on at all) -- this is the
-- third variant, gated on the signed-in caller's own JWT email matching the
-- Request's Contact email, the exact same match get_received_request
-- itself already uses (migration 078). Mints into request_links (migration
-- 075) with the same 30-day expiry, so it's indistinguishable from an
-- owner-minted or cron-minted token to every downstream reader
-- (get_request_by_token, set_response_done_by_token, add_dialog_by_token)
-- -- a Recipient who reaches this link later gets the exact same anonymous,
-- no-sign-in /r/[token] experience RequestResponseForm.tsx already gives a
-- first-time Recipient.
--
-- Granted to authenticated only, never anon -- an anonymous visitor has no
-- JWT email to match against, so there would be nothing for the check to
-- gate on; this function only ever needs to run from ResponseDetailForm.tsx,
-- which is itself RequireAuth-wrapped.
-- ============================================================================

create or replace function public.issue_request_link_as_recipient(p_request_id uuid)
returns text
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_contact_email text;
  v_token text;
begin
  select lower(trim(c.email)) into v_contact_email
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = p_request_id;

  if v_contact_email is null or v_email = '' or v_contact_email <> v_email then
    raise exception 'Not found.';
  end if;

  v_token := encode(gen_random_bytes(32), 'hex');

  insert into public.request_links (request_id, token_hash, expires_at)
    values (p_request_id, digest(v_token, 'sha256'), now() + interval '30 days');

  insert into public.events (actor_user, subject_type, subject_id, request_id, action, detail)
    values (auth.uid(), 'link', p_request_id, p_request_id, 'issued', '{"via": "recipient"}'::jsonb);

  return v_token;
end;
$$;

revoke all on function public.issue_request_link_as_recipient(uuid) from public;
grant execute on function public.issue_request_link_as_recipient(uuid) to authenticated;
