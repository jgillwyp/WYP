-- ============================================================================
-- WYP migration 029 — get_received_print_detail(): full Dialog + Attachments
-- for the Received print report
-- DRAFTED 2026-08-15 — NOT YET CONFIRMED RUN.
--
-- Context: the owner's own xlsx print mockups (Main Screen sections - Sent/
-- Received/ToDos) show each record's full Dialog thread (every Question/
-- Answer/Comment, in full) and full Attachments/Locations list under its
-- description, not just an icon indicating either exists. Sent and ToDos
-- already load dialog/attachments as PostgREST count embeds only
-- (dialog(count), attachments(count) — MainScreen.tsx's own .select()
-- strings), same as Received's own dialog_count/attachment_count (migration
-- 027) — none of the three sections has ever loaded full Dialog/Attachment
-- *content* for the Main Screen list, only counts, since the on-screen rows
-- only ever needed an icon, not the text.
--
-- Deliberately NOT switching the main list queries themselves to load full
-- content by default — that would add real Dialog-thread and Attachment-
-- metadata payload weight to every ordinary Main Screen load, for content
-- the on-screen UI never displays. Instead, the full content is fetched only
-- at the moment Print is actually clicked, for just the ids currently on
-- screen (see MainScreen.tsx's loadPrintDetail/startPrint).
--
-- Sent and ToDos can do this fetch as a plain client-side query — both are
-- rows the signed-in owner already has SELECT access to on public.dialog/
-- public.attachments under existing RLS (migration 025: "SELECT is
-- owner-only"). Received cannot: a Received row's dialog/attachments belong
-- to a different owner, invisible to the recipient's own RLS the same way
-- Category is (CLAUDE.md's Entitlements section) — this function is that
-- gap's fix, parallel to get_received_requests() (migration 012) itself,
-- keyed by the same contacts.email match against the caller's own session
-- email (auth.jwt() ->> 'email'), not a new access model.
--
-- Returns one row per matched request id (silently drops any id in p_ids
-- that isn't actually a Received request for the caller — same "return
-- nothing rather than error" posture as every other token/session-scoped
-- function here), with dialog and attachments each as a jsonb array so one
-- round trip covers every row on the current Received print. Field lists
-- match RequestDetailForm.tsx's own dialog select (id, kind, body, who,
-- replies_to_id) and the attachments API route's own select (id, kind,
-- file_name, reference_url, reference_note) — no new fields invented.
-- ============================================================================

begin;

create or replace function public.get_received_print_detail(p_ids uuid[])
returns table (
  request_id uuid,
  dialog jsonb,
  attachments jsonb
)
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
begin
  if v_email = '' or p_ids is null or array_length(p_ids, 1) is null then
    return; -- no email claim, or nothing asked for
  end if;

  return query
    select
      r.id as request_id,
      coalesce((
        select jsonb_agg(
                 jsonb_build_object(
                   'id', d.id, 'kind', d.kind, 'body', d.body,
                   'who', d.who, 'replies_to_id', d.replies_to_id
                 ) order by d.id
               )
        from public.dialog d
        where d.request_id = r.id
      ), '[]'::jsonb) as dialog,
      coalesce((
        select jsonb_agg(
                 jsonb_build_object(
                   'id', a.id, 'kind', a.kind, 'file_name', a.file_name,
                   'reference_url', a.reference_url, 'reference_note', a.reference_note
                 ) order by a.created_at
               )
        from public.attachments a
        where a.request_id = r.id and a.deleted_at is null
      ), '[]'::jsonb) as attachments
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = any(p_ids)
      and lower(trim(c.email)) = v_email;
end;
$$;

revoke all on function public.get_received_print_detail(uuid[]) from public;
grant execute on function public.get_received_print_detail(uuid[]) to authenticated;

commit;

-- WYP migration 030 — get_contact_request_counts(): per-contact Sent/Rec'd
-- counts for the new Contacts print report.
-- DRAFTED 2026-08-15 — CONFIRMED RUN by the owner 2026-08-15.
--
-- Owner, asked to pick between two readings of "Sent"/"Rec'd" on his own
-- Contacts print mockup: "Sent = requests I sent them; Rec'd = requests
-- they sent me" — the same vocabulary the app's own Main Screen Sent/
-- Received sections already use, not "how many of my Sent requests to them
-- are Done."
--
-- Sent is trivial and needs no new privilege — a plain count of my own
-- requests grouped by contact_id, already fully covered by existing RLS
-- (requests: owner-only). Rec'd is the hard half: get_received_requests()
-- (migration 012) already knows how to match "a Request sent to me" via
-- contacts.email on the *sender's* own contacts table equaling my session
-- email, but it never needed to know the sender's own real account email,
-- only their display name (owner_name). To attribute a Received Request
-- back to *which one of my own contacts* sent it, this function looks up
-- the sender's auth.users.email (their actual login email, not a display
-- name) and matches it against my own contacts.email. Reads auth.users
-- directly inside a SECURITY DEFINER function — same precedent and
-- reasoning as can_create_account() (migration 015): only this function's
-- owner role can see auth.users; a caller only ever gets integer counts
-- back, never anyone's raw email address.
--
-- Returns one row per one of the caller's own contacts (zero-filled counts
-- where there's no match), scoped entirely to the caller via auth.uid()/
-- auth.jwt() — nothing here is reachable by anon.
-- ============================================================================

begin;

create or replace function public.get_contact_request_counts()
returns table (
  contact_id uuid,
  sent_count bigint,
  received_count bigint
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_uid uuid := auth.uid();
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
begin
  if v_uid is null then
    return; -- no session; nothing to compute
  end if;

  return query
    select
      c.id as contact_id,
      coalesce(sent.cnt, 0) as sent_count,
      coalesce(recv.cnt, 0) as received_count
    from public.contacts c
    left join (
      select r.contact_id, count(*) as cnt
      from public.requests r
      where r.owner_id = v_uid
      group by r.contact_id
    ) sent on sent.contact_id = c.id
    left join (
      -- Every Request actually received by me, then rolled up by the real
      -- sender email — mirrors get_received_requests()'s own match, plus
      -- one extra join to auth.users to get an email worth matching against
      -- my own contacts.email (owner_name/profiles alone isn't unique or
      -- guaranteed to match how I saved this person).
      select lower(trim(au.email)) as sender_email, count(*) as cnt
      from public.requests r2
      join public.contacts c2 on c2.id = r2.contact_id
      join auth.users au on au.id = r2.owner_id
      where v_email <> '' and lower(trim(c2.email)) = v_email
      group by lower(trim(au.email))
    ) recv on recv.sender_email = lower(trim(c.email))
    where c.owner_id = v_uid;
end;
$$;

revoke all on function public.get_contact_request_counts() from public;
grant execute on function public.get_contact_request_counts() to authenticated;

commit;

-- ----------------------------------------------------------------------------
-- get_received_request — add received_archived_at, deferred from migration
-- 028 ("revisit if a future Un-Archive-from-Detail affordance needs it").
-- returns jsonb, so a plain CREATE OR REPLACE is safe (no OUT-parameter row
-- shape to fight, unlike get_received_requests()'s own RETURNS TABLE).
-- Byte-identical to the migration 020 version otherwise.
-- ----------------------------------------------------------------------------
create or replace function public.get_received_request(p_request_id uuid)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.received_archived_at,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         c.email as contact_email
    into v_request
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    left join public.profiles p on p.id = r.owner_id
    where r.id = p_request_id;

  if v_request.id is null or v_email = '' or lower(trim(v_request.contact_email)) <> v_email then
    raise exception 'Request not found.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('request', v_request.id, v_request.id, 'viewed_by_recipient', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'received_archived_at', v_request.received_archived_at,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_received_request(uuid) from public;
grant execute on function public.get_received_request(uuid) to authenticated;

-- ----------------------------------------------------------------------------
-- set_response_done_as_recipient — clear received_archived_at in the same
-- UPDATE whenever p_done_date is null (un-archive-on-clear); preserved
-- unchanged otherwise. Byte-identical to the migration 012 version apart
-- from that one added case expression.
-- ----------------------------------------------------------------------------
create or replace function public.set_response_done_as_recipient(
  p_request_id uuid,
  p_done_date date,
  p_done_time time
)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request_id uuid;
begin
  select r.id into v_request_id
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = p_request_id
      and v_email <> ''
      and lower(trim(c.email)) = v_email;

  if v_request_id is null then
    raise exception 'Request not found.';
  end if;

  update public.requests
    set done_date = p_done_date,
        done_time = p_done_time,
        received_archived_at = case when p_done_date is null then null else received_archived_at end
    where id = v_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values (
      'request', v_request_id, v_request_id, 'response_done_updated',
      jsonb_build_object('done_date', p_done_date, 'done_time', p_done_time, 'via', 'recipient_session')
    );
end;
$$;

revoke all on function public.set_response_done_as_recipient(uuid, date, time) from public;
grant execute on function public.set_response_done_as_recipient(uuid, date, time) to authenticated;

-- ============================================================================
-- WYP migration 033 — cron_issue_request_link(): service_role-only response
-- link minting for the cron notification system.
-- DRAFTED 2026-08-17 — NOT YET CONFIRMED RUN.
--
-- app/api/cron/tick/route.ts (built alongside this migration) needs to embed
-- a working /r/[token] response link in every Reminder and Overdue email it
-- sends — the same link Request Detail's own "Get Response Link" band used
-- to expose manually (removed 2026-08-14) and CreateRequestForm.tsx already
-- mints automatically on Send. That existing function, issue_request_link
-- (migration 008), is owner-only by design: it checks auth.uid() = the
-- Request's own owner_id. A cron run has no session — auth.uid() is null —
-- so it cannot call issue_request_link at all, and must not: relaxing that
-- function's own check would let any authenticated caller mint a link for
-- any Request, a real privilege escalation.
--
-- cron_issue_request_link is a separate function, not a relaxed version of
-- issue_request_link — same token/hash/expiry mechanics (30-day expiry,
-- salted SHA-256 hash, existing events logging), but with NO auth.uid()
-- check at all. Safe only because it is never granted to authenticated or
-- anon — only to service_role, which app/api/cron/tick/route.ts alone holds
-- (SUPABASE_SERVICE_ROLE_KEY, server-only env var, same posture as
-- app/api/attachments/*/route.ts). A caller who already has service_role
-- can bypass RLS and read/write the requests table directly regardless, so
-- this function grants no new capability to anyone who didn't already have
-- it — it exists only so the cron route doesn't have to reimplement
-- token-hash generation in JS and risk a byte-level mismatch against
-- get_request_by_token's own `digest(p_token, 'sha256')` comparison.
--
-- Always mints a fresh token and overwrites link_token_hash, exactly like
-- issue_request_link itself already does unconditionally on every call
-- (migration 008 has no "reuse if still valid" branch either) — deliberate,
-- not an oversight. Only the hash is ever persisted (CLAUDE.md's own
-- "store digest(token,'sha256') not the token" rule), so a still-valid
-- earlier link's raw value can never be recovered to re-embed in a second
-- email; the only options are mint-fresh-and-supersede (what this does, and
-- what the app has always done) or leave a stale link in an older email
-- forever un-refreshed. A recipient who already clicked an earlier email's
-- link and is mid-session on /r/[token] is unaffected either way — that
-- page holds the token client-side for the rest of its own visit,
-- independent of what the database's own current hash is.
-- ============================================================================

begin;

create or replace function public.cron_issue_request_link(p_request_id uuid)
returns text
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_owner uuid;
  v_token text;
begin
  select owner_id into v_owner from public.requests where id = p_request_id;

  if v_owner is null then
    raise exception 'Not found.';
  end if;

  v_token := encode(gen_random_bytes(32), 'hex');

  update public.requests
    set link_token_hash = digest(v_token, 'sha256'),
        link_expires_at = now() + interval '30 days',
        link_revoked_at = null
    where id = p_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', p_request_id, p_request_id, 'issued', '{"via": "cron"}'::jsonb);

  return v_token;
end;
$$;

create or replace function public.can_toggle_tier()
returns boolean
language plpgsql
security definer
set search_path = public, auth
as $$
declare
  v_gate_enabled boolean;
  v_email text;
  v_allowlisted boolean;
begin
  select coalesce(
    (select value from app_settings where key = 'tier_toggle_gate_enabled'),
    true
  ) into v_gate_enabled;

  if not v_gate_enabled then
    return true;
  end if;

  select lower(trim(email)) into v_email from auth.users where id = auth.uid();

  if v_email is null then
    return false;
  end if;

  select exists(
    select 1 from tier_toggle_allowlist where lower(email) = v_email
  ) into v_allowlisted;

  return v_allowlisted;
end;
$$;

revoke all on function public.can_toggle_tier() from public;
grant execute on function public.can_toggle_tier() to authenticated;

create or replace function public.set_tier_for_testing(p_tier text)
returns void

security definer
set search_path = public, auth
as $$

  update public.profiles
    set tier = p_tier

end;
$$;

create or replace function public.get_request_by_token(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.reminder_enabled,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         ct.display_name as contact_name,
         r.link_expires_at, r.link_revoked_at
    into v_request
    from public.requests r
    left join public.profiles p on p.id = r.owner_id
    left join public.contacts ct on ct.id = r.contact_id
    where r.link_token_hash = v_hash;

  if v_request.id is null
     or v_request.link_revoked_at is not null
     or v_request.link_expires_at < now()
  then
    raise exception 'This link is no longer valid.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', v_request.id, v_request.id, 'viewed', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'reminder_enabled', v_request.reminder_enabled,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'contact_name', v_request.contact_name,
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_request_by_token(text) from public;
grant execute on function public.get_request_by_token(text) to anon, authenticated;

create or replace function public.set_response_done_as_recipient(
  p_request_id uuid,
  p_done_date date,
  p_done_time time,
  p_reminder_enabled boolean default null
)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request_id uuid;

  select r.id into v_request_id
    from public.requests r
    join public.contacts c on c.id = r.contact_id

      'request', v_request_id, v_request_id, 'response_done_updated',
      jsonb_build_object('done_date', p_done_date, 'done_time', p_done_time, 'reminder_enabled', p_reminder_enabled, 'via', 'recipient_session')
    );
end;

-- ============================================================================
-- WYP migration 038 — Repeat feature (recurring Requests and ToDos)
-- DRAFTED 2026-08-21 — CONFIRMED RUN by the owner 2026-08-21.
--
-- Jim's own recurrence-method design ("WYP Repeat design.docx", uploaded
-- 2026-08-21) — a Request or ToDo can repeat on a Day/Week/Month/Year
-- schedule, anchored to its own Due Date (never Done Date — Jim's explicit
-- instruction: "the Due Date should be the determinant as to the generation
-- of the next Request"). See app/src/lib/repeatRule.ts for the pure
-- recurrence math and descriptive-text builder shared by every screen.
--
-- requests.repeat_rule (jsonb) — the whole RepeatRule object (type,
-- interval, weekdaysOnly, monthMode, stopType, stopDate, stopCount — see
-- repeatRule.ts's own RepeatRule type for the exact shape). Null means "not
-- repeating." Copied verbatim onto every generated successor so the chain
-- continues without re-reading the original.
--
-- requests.repeat_occurrence_index (integer) — 1-based position in the
-- series; the very first Request/ToDo (the one the user configured Repeat
-- on) is 1, its first generated successor is 2, and so on. Only meaningful
-- when repeat_rule is set; needed for the "stops after N occurrences"
-- condition, which can't be answered by counting rows — nothing links a
-- series together beyond each row's own repeat_rule copy (no series id,
-- since nothing in this feature needs to list a whole series, only ever
-- the current and next occurrence).
--
-- requests.repeat_next_generated_at (timestamptz) — idempotency marker, same
-- convention as reminder_sent_at/overdue_notified_at (migration 032): set by
-- the cron job (app/api/cron/tick/route.ts, new Phase E) the moment it has
-- processed this row's own Due-Date-arrival for repeat-generation purposes,
-- whether that produced a successor or hit a stop condition. Prevents a
-- second cron run in the same hour from double-inserting.
--
-- attachments.carry_into_repeats (boolean) — Jim's own design for carrying
-- Attachments/Locations into a repeating series: at Send/Save time (or
-- later, via Edit Repeat), the sender is shown the existing Attachments/
-- Locations list with checkboxes and told Dialog is never carried forward,
-- only whichever items they check. A boolean on the row itself, not a
-- separate join table or an array column on requests — the cron job's own
-- generation step just selects `where carry_into_repeats = true` on the
-- occurrence it's about to supersede, and stamps the same flag on every
-- duplicate it creates, so the selection propagates down the whole chain
-- with no re-asking.
--
-- Real 'file' Attachments are duplicated as a fresh Storage object per
-- generated occurrence (app/api/cron/tick/route.ts uses storage.copy()),
-- never shared across multiple `attachments` rows — matches this table's
-- existing hard-delete-per-row model (migration 025's own header comment)
-- and avoids the complexity of reference-counted deletes. Accepted
-- trade-off, confirmed with Jim: an unbounded "every day, stops never"
-- repeat with a carried file keeps duplicating that file's bytes into every
-- occurrence indefinitely — a real but currently acceptable storage cost at
-- this app's testing scale.
--
-- Archived Requests/ToDos are exempt from repeat generation, same posture
-- CLAUDE.md/the cron system already takes for every other notification
-- (migration 032's own "archived items are exempt from every notification"
-- rule) — an archived row's Repeat control is already greyed/non-modifiable
-- in the UI (Jim's own spec), so the cron job honors that by simply never
-- generating a successor for an archived row, rather than generating one
-- silently behind the scenes.
-- ============================================================================

begin;

alter table public.requests
  add column if not exists repeat_rule jsonb,
  add column if not exists repeat_occurrence_index integer,
  add column if not exists repeat_next_generated_at timestamptz;

comment on column public.requests.repeat_rule is
  'Recurrence rule (Day/Week/Month/Year + interval + stop condition), or null if this Request/ToDo does not repeat. Shape defined by RepeatRule in app/src/lib/repeatRule.ts. Copied verbatim onto every generated successor.';
comment on column public.requests.repeat_occurrence_index is
  '1-based position in this row''s own repeat series (1 = the original, 2 = its first generated successor, ...). Only meaningful when repeat_rule is set.';
comment on column public.requests.repeat_next_generated_at is
  'Idempotency marker for the cron job''s Repeat-generation phase — set once this row''s own Due Date has been processed (successor generated, or a stop condition was hit), whether or not a new row was actually inserted.';

alter table public.attachments
  add column if not exists carry_into_repeats boolean not null default false;

comment on column public.attachments.carry_into_repeats is
  'Set at Send/Save time (or later via Edit Repeat) when the sender selects this Attachment/Location to carry into every future occurrence of a repeating Request/ToDo. The cron job duplicates every row with this flag set onto each newly generated occurrence, propagating the flag onto the duplicate too, so the selection continues down the whole chain without re-asking.';

-- repeat_rule/repeat_occurrence_index/repeat_next_generated_at need no new
-- RLS/grants — plain requests columns, already covered by that table's
-- existing owner-scoped policies (migration 002/003).
--
-- carry_into_repeats is different: migration 025 deliberately created NO
-- update policy on attachments at all ("an attachment is added or removed,
-- never edited in place" — that migration's own header comment), so a
-- direct client toggle of this new column would silently fail RLS with no
-- policy to admit it. Rather than open a general UPDATE (which would
-- contradict that same append-only design), this adds a narrow one scoped
-- to a single column — same two-part shape (RLS policy + column-level
-- GRANT) migration 013 already established for profiles.time_zone: the RLS
-- policy admits an UPDATE on any row belonging to the caller's own
-- Request/ToDo, and the GRANT then restricts that permission to touching
-- only carry_into_repeats — an owner still cannot rename, resize, or
-- otherwise edit an existing attachment in place.
create policy "attachments: owners update carry_into_repeats"
  on public.attachments for update to authenticated
  using (request_id in (select id from public.requests where owner_id = auth.uid()))
  with check (request_id in (select id from public.requests where owner_id = auth.uid()));

grant update (carry_into_repeats) on public.attachments to authenticated;

commit;

-- ----------------------------------------------------------------------------
-- get_request_by_token — add reminder_day_of_enabled/reminder_day_of_sent_at
-- to the jsonb payload.
-- ----------------------------------------------------------------------------
create or replace function public.get_request_by_token(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.reminder_enabled, r.reminder_sent_at, r.overdue_reminder_enabled, r.reminder_day_of_enabled,
         r.reminder_day_of_sent_at, r.repeat_rule,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         ct.display_name as contact_name,
         r.link_expires_at, r.link_revoked_at
    into v_request
    from public.requests r
    left join public.profiles p on p.id = r.owner_id
    left join public.contacts ct on ct.id = r.contact_id
    where r.link_token_hash = v_hash;

  if v_request.id is null
     or v_request.link_revoked_at is not null
     or v_request.link_expires_at < now()
  then
    raise exception 'This link is no longer valid.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', v_request.id, v_request.id, 'viewed', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'reminder_enabled', v_request.reminder_enabled,
    'reminder_sent_at', v_request.reminder_sent_at,
    'overdue_reminder_enabled', v_request.overdue_reminder_enabled,
    'reminder_day_of_enabled', v_request.reminder_day_of_enabled,
    'reminder_day_of_sent_at', v_request.reminder_day_of_sent_at,
    'repeat_rule', v_request.repeat_rule,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'contact_name', v_request.contact_name,
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_request_by_token(text) from public;
grant execute on function public.get_request_by_token(text) to anon, authenticated;

-- ----------------------------------------------------------------------------
-- get_received_request — same addition, signed-in recipient path.
-- ----------------------------------------------------------------------------
create or replace function public.get_received_request(p_request_id uuid)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.received_archived_at, r.reminder_enabled, r.reminder_sent_at, r.overdue_reminder_enabled,
         r.reminder_day_of_enabled, r.reminder_day_of_sent_at, r.repeat_rule,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         c.email as contact_email
    into v_request
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    left join public.profiles p on p.id = r.owner_id
    where r.id = p_request_id;

  if v_request.id is null or v_email = '' or lower(trim(v_request.contact_email)) <> v_email then
    raise exception 'Request not found.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('request', v_request.id, v_request.id, 'viewed_by_recipient', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'received_archived_at', v_request.received_archived_at,
    'reminder_enabled', v_request.reminder_enabled,
    'reminder_sent_at', v_request.reminder_sent_at,
    'overdue_reminder_enabled', v_request.overdue_reminder_enabled,
    'reminder_day_of_enabled', v_request.reminder_day_of_enabled,
    'reminder_day_of_sent_at', v_request.reminder_day_of_sent_at,
    'repeat_rule', v_request.repeat_rule,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_received_request(uuid) from public;
grant execute on function public.get_received_request(uuid) to authenticated;

-- ----------------------------------------------------------------------------
-- set_response_done_by_token — gains p_reminder_day_of_enabled, same
-- coalesce-against-existing-column pattern as p_reminder_enabled/
-- p_overdue_reminder_enabled.
-- ----------------------------------------------------------------------------
create or replace function public.set_response_done_by_token(
  p_token text,
  p_done_date date,
  p_done_time time,
  p_reminder_enabled boolean default null,
  p_overdue_reminder_enabled boolean default null,
  p_reminder_day_of_enabled boolean default null
)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request_id uuid;
begin
  select r.id into v_request_id
    from public.requests r
    where r.link_token_hash = v_hash
      and r.link_revoked_at is null
      and r.link_expires_at > now();

  if v_request_id is null then
    raise exception 'This link is no longer valid.';
  end if;

  update public.requests
    set done_date = p_done_date,
        done_time = p_done_time,
        reminder_enabled = coalesce(p_reminder_enabled, reminder_enabled),
        overdue_reminder_enabled = coalesce(p_overdue_reminder_enabled, overdue_reminder_enabled),
        reminder_day_of_enabled = coalesce(p_reminder_day_of_enabled, reminder_day_of_enabled)
    where id = v_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('request', v_request_id, v_request_id, 'response_saved',
      jsonb_build_object('done_date', p_done_date, 'done_time', p_done_time, 'reminder_enabled', p_reminder_enabled, 'overdue_reminder_enabled', p_overdue_reminder_enabled, 'reminder_day_of_enabled', p_reminder_day_of_enabled));
end;
$$;

revoke all on function public.set_response_done_by_token(text, date, time, boolean, boolean, boolean) from public;
grant execute on function public.set_response_done_by_token(text, date, time, boolean, boolean, boolean) to anon, authenticated;

-- ----------------------------------------------------------------------------
-- set_response_done_as_recipient — same addition, signed-in recipient path.
-- ----------------------------------------------------------------------------
create or replace function public.set_response_done_as_recipient(
  p_request_id uuid,
  p_done_date date,
  p_done_time time,
  p_reminder_enabled boolean default null,
  p_overdue_reminder_enabled boolean default null,
  p_reminder_day_of_enabled boolean default null
)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_contact_email text;
begin
  select lower(trim(c.email)) into v_contact_email
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = p_request_id;

  if v_email = '' or v_contact_email is null or v_contact_email <> v_email then
    raise exception 'Request not found.';
  end if;

  update public.requests
    set done_date = p_done_date,
        done_time = p_done_time,
        received_archived_at = case when p_done_date is null then received_archived_at else null end,
        reminder_enabled = coalesce(p_reminder_enabled, reminder_enabled),
        overdue_reminder_enabled = coalesce(p_overdue_reminder_enabled, overdue_reminder_enabled),
        reminder_day_of_enabled = coalesce(p_reminder_day_of_enabled, reminder_day_of_enabled)
    where id = p_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('request', p_request_id, p_request_id, 'response_saved',
      jsonb_build_object('done_date', p_done_date, 'done_time', p_done_time, 'reminder_enabled', p_reminder_enabled, 'overdue_reminder_enabled', p_overdue_reminder_enabled, 'reminder_day_of_enabled', p_reminder_day_of_enabled, 'via', 'recipient_session'));
end;
$$;

revoke all on function public.set_response_done_as_recipient(uuid, date, time, boolean, boolean, boolean) from public;
grant execute on function public.set_response_done_as_recipient(uuid, date, time, boolean, boolean, boolean) to authenticated;

-- ----------------------------------------------------------------------------
-- get_request_by_token — add owner_request_reminders_enabled to the jsonb
-- payload, same coalesce-true-if-null pattern already used for
-- owner_request_time_enabled (a null only happens if the owner's own
-- profiles row is somehow missing, which should never occur in practice).
-- jsonb-returning, so a plain create-or-replace is safe -- no OUT-parameter
-- shape to worry about (see migration 017's own header for why that
-- constraint doesn't apply to jsonb functions).
-- ----------------------------------------------------------------------------
create or replace function public.get_request_by_token(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.reminder_enabled, r.reminder_sent_at, r.overdue_reminder_enabled, r.reminder_day_of_enabled,
         r.reminder_day_of_sent_at, r.repeat_rule,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         p.request_reminders_enabled as owner_request_reminders_enabled,
         ct.display_name as contact_name,
         r.link_expires_at, r.link_revoked_at
    into v_request
    from public.requests r
    left join public.profiles p on p.id = r.owner_id
    left join public.contacts ct on ct.id = r.contact_id
    where r.link_token_hash = v_hash;

  if v_request.id is null
     or v_request.link_revoked_at is not null
     or v_request.link_expires_at < now()
  then
    raise exception 'This link is no longer valid.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', v_request.id, v_request.id, 'viewed', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'reminder_enabled', v_request.reminder_enabled,
    'reminder_sent_at', v_request.reminder_sent_at,
    'overdue_reminder_enabled', v_request.overdue_reminder_enabled,
    'reminder_day_of_enabled', v_request.reminder_day_of_enabled,
    'reminder_day_of_sent_at', v_request.reminder_day_of_sent_at,
    'repeat_rule', v_request.repeat_rule,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'owner_request_reminders_enabled', coalesce(v_request.owner_request_reminders_enabled, true),
    'contact_name', v_request.contact_name,
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_request_by_token(text) from public;
grant execute on function public.get_request_by_token(text) to anon, authenticated;

-- requests.contact_id — kind='reference' already implies "was a Location."

-- Per-request_id entries are joined with ", ", each entry rendered as
-- "note: value" when a Location has both a Description and a Location
-- value, or just whichever of the two is present otherwise — Jim confirmed
-- this format needs no change from the first draft. Requests with no
-- Locations at all are untouched (the loop simply never visits them).
--
-- Runs before the application code deploys that stops writing/reading
-- kind='reference' rows at all — once TodoDetailForm.tsx and
-- CreateTodoForm.tsx switch to mode='file', nothing in the app inserts a
-- 'reference' row again, so deleting every existing one here is a clean,
-- one-time cutover, not an ongoing behavior change.

begin;

do $$
declare
  r record;
  locs text;
begin
  for r in
    select request_id,
           string_agg(
             coalesce(nullif(reference_note, '') || ': ', '') ||
             coalesce(reference_url, reference_note, ''),
             ', ' order by created_at
           ) as locs
    from public.attachments
    where kind = 'reference'
    group by request_id

      update public.requests
        set description = coalesce(description, '') || ' -- Location(s): ' || r.locs

  end loop;

-- WYP migration 049 — reword get_request_by_token's generic dead-link
-- message so it reads as sender action, not app failure
-- CONFIRMED RUN by the owner 2026-09-02.
--
-- Jim, concerned that a recipient clicking an old Request Response email
-- link after the WYP user has deleted the underlying Request (via a Contact
-- deletion cascade, 2026-09-01, or Archive's own standalone Delete,
-- 2026-08-15) could read the resulting dead link as the app being
-- unreliable rather than as something the sender did. Both deletion paths
-- hard-delete the requests row outright — a deliberate choice, see
-- CLAUDE.md's Contact-deletion history — so link_token_hash is gone with
-- it; get_request_by_token has nothing left to distinguish "a real Request
-- the sender removed" from "a malformed or never-valid token" — both hit
-- the same v_request.id is null branch. Confirmed with Jim: don't try to
-- distinguish the cases at the DB level (would mean keeping some tombstone
-- row after every delete, reopening the hard-delete-for-privacy decision)
-- — just change what the one generic message says, since it already covers
-- every failure mode (never-existed, expired, revoked, deleted) alike.
--
-- Wording changes from "This link is no longer valid." to "This link is no
-- longer active. The sender has removed the Request it pointed to." —
-- attributes the dead link to sender action for the common real case
-- (deletion), at the cost of being technically imprecise for the rare
-- genuinely-malformed-link case (a mistyped or truncated URL that was
-- never valid) — an accepted tradeoff, confirmed with Jim rather than
-- assumed. Same message, same reasoning, still covers the token-expiry and
-- link-revocation branches too (all three already shared this one message
-- before this change; still undistinguished after it).
--
-- Body is otherwise byte-for-byte identical to migration 045's version —
-- only the raised exception's message text changes. jsonb-returning, so a
-- plain create-or-replace is safe (see migration 017's own header for why
-- the OUT-parameter constraint doesn't apply here).
--
-- Scoped to get_request_by_token only — the anonymous /r/[token] mailed-
-- link path, the actual scenario Jim described. get_received_request's own
-- "Request not found." message is a different situation (a signed-in
-- recipient reaching a specific Detail URL directly, not a durable mailed
-- link meant to keep working indefinitely) and is left unchanged.
--
-- The app-side fallback strings in RequestResponseForm.tsx (used only if a
-- network-level failure returns no message at all) are updated to match in
-- the same commit as this migration.
-- ============================================================================

create or replace function public.get_request_by_token(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.reminder_enabled, r.reminder_sent_at, r.overdue_reminder_enabled, r.reminder_day_of_enabled,
         r.reminder_day_of_sent_at, r.repeat_rule,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         p.request_reminders_enabled as owner_request_reminders_enabled,
         ct.display_name as contact_name,
         r.link_expires_at, r.link_revoked_at
    into v_request
    from public.requests r
    left join public.profiles p on p.id = r.owner_id
    left join public.contacts ct on ct.id = r.contact_id
    where r.link_token_hash = v_hash;

  if v_request.id is null
     or v_request.link_revoked_at is not null
     or v_request.link_expires_at < now()
  then
    raise exception 'This link is no longer active. The sender has removed the Request it pointed to.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', v_request.id, v_request.id, 'viewed', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'reminder_enabled', v_request.reminder_enabled,
    'reminder_sent_at', v_request.reminder_sent_at,
    'overdue_reminder_enabled', v_request.overdue_reminder_enabled,
    'reminder_day_of_enabled', v_request.reminder_day_of_enabled,
    'reminder_day_of_sent_at', v_request.reminder_day_of_sent_at,
    'repeat_rule', v_request.repeat_rule,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'owner_request_reminders_enabled', coalesce(v_request.owner_request_reminders_enabled, false),
    'contact_name', v_request.contact_name,
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

-- WYP migration 050 — fixes add_dialog_as_recipient's events insert
-- (same uuid/bigint bug migration 010 already fixed once, in a sibling
-- function) — CONFIRMED RUN by the owner, 2026-09-02.
--
-- BUG, owner-reported: a signed-in recipient answering a Question on
-- Response Detail (get_received_request's own write path) failed with
-- `column "subject_id" is of type uuid but expression is of type bigint`.
-- Jim also reported answering his own Question via Request Detail worked
-- fine — expected, since the owner's own screens (Create Request/Request
-- Detail) write Dialog with a plain client-side insert, never through this
-- RPC at all; only the recipient-facing path is affected.
--
-- Root cause is byte-for-byte the same defect migration 010 (Week3 SQL
-- history) already found and fixed in add_dialog_by_token: the events row
-- was logged as ('dialog', v_new_id, ...) — v_new_id is the new dialog
-- row's own id, a plain bigint identity column (same type
-- dialog.replies_to_id already points at) — but events.subject_id is uuid
-- (migration 002) and always has been; no bigint value can ever be cast
-- into it. add_dialog_as_recipient was written at migration 012
-- (2026-08-11, Week4 SQL history), a day AFTER migration 010 already fixed
-- this exact shape of bug in add_dialog_by_token — the fix was never
-- ported across to this sibling function, which reintroduced the identical
-- defect. get_request_by_token, set_response_done_by_token, and
-- set_response_done_as_recipient all already log with the Request's own
-- uuid as subject_id, so none of them ever hit this; add_dialog_by_token
-- and add_dialog_as_recipient are the only two functions that ever insert
-- a *new* dialog row and log it, which is exactly the shape that trips the
-- type mismatch.
--
-- FIX: identical to migration 010's own fix — subject_id = v_request_id
-- (the Request's own uuid, already available in this function), not the
-- new dialog row's bigint id. The dialog id is preserved inside detail
-- instead (dialog_id), same convention as every other events insert in
-- this function family.
--
-- Body is otherwise byte-for-byte identical to the version in Week4 SQL
-- history (migration 012) — only the events insert changes.
-- ----------------------------------------------------------------------------

create or replace function public.add_dialog_as_recipient(
  p_request_id uuid,
  p_kind text,
  p_body text,
  p_replies_to_id bigint default null
)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request_id uuid;
  v_who text;
  v_new_id bigint;
  v_created_at timestamptz;
begin
  select r.id into v_request_id
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = p_request_id
      and v_email <> ''
      and lower(trim(c.email)) = v_email;

  if v_request_id is null then
    raise exception 'Request not found.';
  end if;

  if p_kind not in ('question', 'answer', 'comment') then
    raise exception 'Invalid Dialog entry type.';
  end if;

  if trim(coalesce(p_body, '')) = '' then
    raise exception 'Enter Dialog Text.';
  end if;

  select coalesce(p.display_name, v_email) into v_who
    from public.profiles p where p.id = auth.uid();
  v_who := coalesce(v_who, v_email, 'Recipient'); -- defensive only

  insert into public.dialog (request_id, kind, body, who, author_user_id, replies_to_id)
    values (
      v_request_id, p_kind, p_body, v_who, auth.uid(),
      case when p_kind = 'answer' then p_replies_to_id else null end
    )
    returning id, created_at into v_new_id, v_created_at;

  -- FIXED (migration 050): subject_id is the Request's own uuid, not the
  -- new dialog row's bigint id — events.subject_id cannot hold a bigint.
  -- The dialog id is still recorded, inside detail.
  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values (
      'dialog', v_request_id, v_request_id, 'created_by_recipient',
      jsonb_build_object('kind', p_kind, 'dialog_id', v_new_id)
    );

  return jsonb_build_object('id', v_new_id, 'created_at', v_created_at, 'who', v_who);
end;
$$;

revoke all on function public.add_dialog_as_recipient(uuid, text, text, bigint) from public;

create or replace function public.set_storage_limit_override(p_bytes bigint)
returns void
language plpgsql
security definer
set search_path = public, auth
as $$
begin
  if not public.can_toggle_tier() then

    set storage_limit_override_bytes = p_bytes
    where id = auth.uid();
end;

-- Migration 054 (2026-09-07) — log_event(): the client-callable half of
-- Admin Statistics' events instrumentation
--
-- Per docs/WYP_Admin_Statistics_Plan.md's "Data foundation" section: public.
-- events (migration 002) is extended to also cover owner-side activity, not
-- just the recipient/secure-link flow it already logs. Several of the
-- "cheap" sites the plan calls out are plain client-side .insert()/.update()
-- calls under normal RLS (AddContactForm.tsx, CreateTodoForm.tsx,
-- TodoDetailForm.tsx) with no existing server code path to hook a write
-- into — unlike the delete/notification routes, which already run
-- server-side and get a service_role events insert added directly in this
-- same batch (see the application-code changes alongside this migration;
-- no SQL needed for those).
--
-- log_event() is that missing server-side hook for the browser-originated
-- sites: one narrow, ownership-checked RPC rather than granting
-- authenticated any direct insert on events (which the table's own
-- 2026-02 comment explicitly rules out — "Nothing writes here from the
-- browser"). A forged action/detail here is cosmetic only, never a
-- privilege escalation: the function insists the row being logged about
-- (a Contact or a Request/ToDo) is actually owned by the caller before
-- writing anything, exactly like every RLS policy already governing those
-- tables, so the worst a malicious authenticated caller could do is log a
-- misleading fact about their own data -- which nothing except the admin
-- statistics screens (themselves re-checking is_admin, migration 053) will
-- ever read.
--
-- p_subject_type is deliberately narrower than events.subject_type's own
-- check constraint ('contact' or 'request' — 'todo' rows live in the same
-- requests table and are told apart the same way the rest of the app
-- already does, by contact_id being null) — dialog/attachment/profile/link
-- events are all still written exclusively by their own existing
-- SECURITY DEFINER functions, untouched by this migration.
-- ============================================================================

create or replace function public.log_event(
  p_subject_type text,
  p_subject_id uuid,
  p_action text,
  p_detail jsonb default '{}'::jsonb
)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_request_id uuid;
begin
  if p_subject_type = 'contact' then
    if not exists (
      select 1 from public.contacts
      where id = p_subject_id and owner_id = auth.uid()
    ) then
      raise exception 'Not found.';
    end if;
    v_request_id := null;
  elsif p_subject_type in ('request', 'todo') then
    if not exists (
      select 1 from public.requests
      where id = p_subject_id and owner_id = auth.uid()
    ) then
      raise exception 'Not found.';
    end if;
    v_request_id := p_subject_id;
  else
    raise exception 'Invalid subject_type for log_event.';
  end if;

  insert into public.events (actor_user, subject_type, subject_id, request_id, action, detail)
    values (auth.uid(), p_subject_type, p_subject_id, v_request_id, p_action, coalesce(p_detail, '{}'::jsonb));
end;
$$;

revoke all on function public.log_event(text, uuid, text, jsonb) from public;
grant execute on function public.log_event(text, uuid, text, jsonb) to authenticated;

-- Not granted to anon — every current call site (Contact created, ToDo
-- created, ToDo changed, ToDo Done) is a signed-in owner action. The
-- recipient/anonymous side keeps writing events through its own existing
-- token-verified functions (set_response_done_by_token, add_dialog_by_token,
-- etc.), which already run this exact ownership-equivalent check inline
-- against the link token instead of auth.uid().

-- Migration 055 (2026-09-07) — set_archived(): owner-side Archive/UnArchive
-- moves to a SECURITY DEFINER function, for Admin Statistics logging
--
-- Per docs/WYP_Admin_Statistics_Plan.md's Task 3: ArchiveForm.tsx's Sent/
-- ToDo Archive and UnArchive actions (Received is untouched by this
-- migration — archive_received_request()/unarchive_received_request(),
-- migrations 028/046, already exist and stay exactly as they are; the plan
-- explicitly scopes Archived/Unarchived tracking to Sent-only, since a
-- Received item isn't owned by this account) have, until now, been a plain
-- client-side `.update({ archived_at })` under normal RLS — the one site
-- in the whole app with no server code path at all to hook an events write
-- into. Every other "cheap" instrumentation site (migration 054) was one
-- more line inside code that already ran server-side; this one genuinely
-- needed new server code, flagged as such in the plan from the start.
--
-- Batched (uuid[]) rather than one-id-at-a-time, matching ArchiveForm.tsx's
-- own existing "check several rows, act once" flow exactly (today it does
-- one .update().in('id', ids) call for the whole selection) — one round
-- trip and one events insert per batch, not N.
--
-- Ownership is enforced the same way "requests: owners update own" (RLS)
-- already would have: `where owner_id = auth.uid()` silently drops any id
-- in the batch the caller doesn't own, rather than raising — the same
-- quiet no-op RLS itself already produced on this exact call before today.
-- ============================================================================

create or replace function public.set_archived(p_request_ids uuid[], p_archived boolean)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_now timestamptz := now();
begin
  with updated as (
    update public.requests
      set archived_at = case when p_archived then v_now else null end
      where id = any(p_request_ids) and owner_id = auth.uid()
      returning id, contact_id
  )
  insert into public.events (actor_user, subject_type, subject_id, request_id, action, detail)
    select
      auth.uid(),
      case when contact_id is not null then 'request' else 'todo' end,
      id,
      id,
      case when p_archived then 'archived' else 'unarchived' end,
      '{}'::jsonb
    from updated;
end;
$$;

revoke all on function public.set_archived(uuid[], boolean) from public;
grant execute on function public.set_archived(uuid[], boolean) to authenticated;

-- Migration 056 (2026-09-07) — admin_stats_contacts(): the Contacts
-- Activity screen's aggregation RPC
--
-- Per docs/WYP_Admin_Statistics_Plan.md's "Contacts — Activity" screen and
-- shared filter row (From/To month range, Weekly Sun-Sat / Monthly
-- granularity, cohort: All / Beta allowlist / one profile). is_admin is
-- re-checked here (not just trusted from the Statistics chip being
-- visible) — same "assume the UI gate was bypassed" posture migration
-- 053's own comment already established for every admin RPC.
--
-- Added/Deleted come from events (migration 054/002) — the only way to
-- count a Contact that has SINCE been hard-deleted; a live table scan of
-- public.contacts can never see it. With Phone/With Notes are read off the
-- SAME 'created' events' own detail jsonb (has_phone/has_notes, captured
-- at creation time by AddContactForm.tsx as of this migration) rather than
-- the current contacts row, so a later edit never rewrites what a past
-- period showed, and a Contact added and deleted inside one period still
-- counts correctly. All four blank out before 2026-09-07 (when this
-- logging began), per the plan's own "no retroactive estimation" decision
-- -- there is no historical detail to read for events that were never
-- written.
--
-- Total is different from the other four: a cumulative running count, not
-- a per-period flow, so it CAN be read straight from the live contacts
-- table (created_at survives independently of events) -- but only for
-- Contacts that still exist today. A Contact created and later deleted
-- disappears from every period's Total from that point on, not just the
-- period it was deleted in -- Total is best read as "how many of today's
-- surviving Contacts existed by then," not "how many existed at that
-- historical moment." This is the same permanent limitation the plan
-- already calls out for Overdue, just on the Contacts screen instead.
--
-- Sunday-start week buckets: Postgres date_trunc('week', ...) is
-- Monday-anchored; shifting the input forward one day before truncating,
-- then subtracting a day back, is the standard trick to re-anchor it to
-- Sunday. period_end is clipped to p_to so the final, possibly-partial
-- bucket's Total never counts Contacts dated after the query's own To
-- date.
--
-- The per-period Total subquery re-scans public.contacts once per row
-- (N periods, not N^2 over the whole date range) -- entirely fine at
-- Private Testing's current scale; worth revisiting only if the Contacts
-- table or the selected date range both grow very large.
-- ============================================================================

create or replace function public.admin_stats_contacts(
  p_from date,
  p_to date,
  p_granularity text,
  p_cohort text,
  p_profile_id uuid default null
)
returns table (
  period_start date,
  period_end date,
  added bigint,
  deleted bigint,
  with_phone bigint,
  with_notes bigint,
  total bigint
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_is_admin boolean;
begin
  select is_admin into v_is_admin from public.profiles where id = auth.uid();
  if not coalesce(v_is_admin, false) then
    raise exception 'Not authorized.';
  end if;
  if p_granularity not in ('week', 'month') then
    raise exception 'Invalid granularity.';
  end if;
  if p_cohort not in ('all', 'beta', 'profile') then
    raise exception 'Invalid cohort.';
  end if;

  return query
  with cohort as (
    select u.id
    from auth.users u
    where p_cohort = 'all'
       or (p_cohort = 'profile' and u.id = p_profile_id)
       or (p_cohort = 'beta' and lower(u.email) in (select lower(email) from public.beta_allowlist))
  ),
  periods as (
    select
      gs::date as period_start,
      (case when p_granularity = 'week'
         then (gs + interval '6 days')
         else (date_trunc('month', gs) + interval '1 month' - interval '1 day')
       end)::date as period_end_natural
    from generate_series(
      case when p_granularity = 'week'
        then date_trunc('week', p_from::timestamp + interval '1 day') - interval '1 day'
        else date_trunc('month', p_from::timestamp)
      end,
      p_to::timestamp,
      case when p_granularity = 'week' then interval '7 days' else interval '1 month' end
    ) as gs
  ),
  bucketed as (
    select p.period_start, least(p.period_end_natural, p_to) as period_end
    from periods p
  ),
  ev_created as (
    select
      (case when p_granularity = 'week'
         then (date_trunc('week', e.at + interval '1 day') - interval '1 day')
         else date_trunc('month', e.at)
       end)::date as period_start,
      count(*) as added,
      count(*) filter (where (e.detail->>'has_phone')::boolean is true) as with_phone,
      count(*) filter (where (e.detail->>'has_notes')::boolean is true) as with_notes
    from public.events e
    where e.subject_type = 'contact' and e.action = 'created'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  ev_deleted as (
    select
      (case when p_granularity = 'week'
         then (date_trunc('week', e.at + interval '1 day') - interval '1 day')
         else date_trunc('month', e.at)
       end)::date as period_start,
      count(*) as deleted
    from public.events e
    where e.subject_type = 'contact' and e.action = 'deleted'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  )
  select
    b.period_start,
    b.period_end,
    coalesce(c.added, 0) as added,
    coalesce(dl.deleted, 0) as deleted,
    coalesce(c.with_phone, 0) as with_phone,
    coalesce(c.with_notes, 0) as with_notes,
    (
      select count(*) from public.contacts ct
      where ct.owner_id in (select id from cohort)
        and ct.created_at::date <= b.period_end
    ) as total
  from bucketed b
  left join ev_created c on c.period_start = b.period_start
  left join ev_deleted dl on dl.period_start = b.period_start
  order by b.period_start;
end;
$$;

revoke all on function public.admin_stats_contacts(date, date, text, text, uuid) from public;
grant execute on function public.admin_stats_contacts(date, date, text, text, uuid) to authenticated;

--

-- (set_response_done_by_token, migration 003) — that fires on every
-- token-based response save regardless of whether Done Date actually
-- changed, which would double-count against this migration's own 'done'
-- events (send-request-update-to-owner logs 'done' only when Done Date is
-- genuinely a changed field and newly non-null). This RPC only ever reads
-- the specific action vocabulary migration 054/055 introduced.
--

-- has no delete path anywhere in this app (append-only by design), so

-- have. author_user_id = the request's own owner_id identifies a Sent
-- (owner-authored) entry directly; anything else (a signed-in recipient's

-- via the same contact-email join.

    select p.period_start, least(p.period_end_natural, p_to) as period_end
    from periods p
  ),
  -- Sent Created: 'created' events, subject_type 'request', by an owner in cohort.
  sent_created_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'request' and e.action = 'created'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  -- Received Created: any 'created' event whose Request's Contact email matches a cohort account.
  received_created_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    join public.requests r on r.id = e.request_id
    join public.contacts c on c.id = r.contact_id
    join auth.users u on lower(u.email) = lower(trim(c.email))
    where e.subject_type = 'request' and e.action = 'created'
      and u.id in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  sent_changed_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'request' and e.action = 'changed'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  received_changed_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    join public.requests r on r.id = e.request_id
    join public.contacts c on c.id = r.contact_id
    join auth.users u on lower(u.email) = lower(trim(c.email))
    where e.subject_type = 'request' and e.action = 'changed' and e.actor_user is null
      and u.id in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  sent_done_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'request' and e.action = 'done'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  received_done_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    join public.requests r on r.id = e.request_id
    join public.contacts c on c.id = r.contact_id
    join auth.users u on lower(u.email) = lower(trim(c.email))
    where e.subject_type = 'request' and e.action = 'done' and e.actor_user is null
      and u.id in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  deleted_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n

    group by 1
  ),
  archived_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,

      and e.at::date between p_from and p_to

  unarchived_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,

  received_attachments_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n

    join public.contacts c on c.id = r.contact_id
    join auth.users u on lower(u.email) = lower(trim(c.email))
    where e.subject_type = 'attachment' and e.action = 'created'

      and u.id in (select id from cohort)

    group by 1
  ),
  sent_dialog_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', d.created_at + interval '1 day') - interval '1 day') else date_trunc('month', d.created_at) end)::date as period_start,
      count(*) as n
    from public.dialog d
    join public.requests r on r.id = d.request_id
    where r.contact_id is not null
      and d.author_user_id = r.owner_id
      and r.owner_id in (select id from cohort)
      and d.created_at::date between p_from and p_to
    group by 1
  ),
  received_dialog_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', d.created_at + interval '1 day') - interval '1 day') else date_trunc('month', d.created_at) end)::date as period_start,
      count(*) as n
    from public.dialog d
    join public.requests r on r.id = d.request_id
    join public.contacts c on c.id = r.contact_id
    join auth.users u on lower(u.email) = lower(trim(c.email))
    where r.contact_id is not null
      and d.author_user_id is distinct from r.owner_id
      and u.id in (select id from cohort)
      and d.created_at::date between p_from and p_to
    group by 1
  )
  select
    b.period_start,
    b.period_end,
    coalesce(sc.n, 0), coalesce(rc.n, 0),
    coalesce(sch.n, 0), coalesce(rch.n, 0),
    coalesce(sd.n, 0), coalesce(rd.n, 0),
    coalesce(del.n, 0),
    coalesce(arc.n, 0),
    coalesce(unarc.n, 0),
    coalesce(sa.n, 0), coalesce(ra.n, 0),
    coalesce(sdlg.n, 0), coalesce(rdlg.n, 0)
  from bucketed b
  left join sent_created_t sc on sc.period_start = b.period_start
  left join received_created_t rc on rc.period_start = b.period_start
  left join sent_changed_t sch on sch.period_start = b.period_start
  left join received_changed_t rch on rch.period_start = b.period_start
  left join sent_done_t sd on sd.period_start = b.period_start
  left join received_done_t rd on rd.period_start = b.period_start
  left join deleted_t del on del.period_start = b.period_start
  left join archived_t arc on arc.period_start = b.period_start
  left join unarchived_t unarc on unarc.period_start = b.period_start
  left join sent_attachments_t sa on sa.period_start = b.period_start
  left join received_attachments_t ra on ra.period_start = b.period_start
  left join sent_dialog_t sdlg on sdlg.period_start = b.period_start
  left join received_dialog_t rdlg on rdlg.period_start = b.period_start
  order by b.period_start;
end;
$$;

  v_is_admin boolean;

    where e.subject_type = 'todo' and e.action = 'done'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  deleted_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'todo' and e.action = 'deleted'

  ),
  attachments_t as (

  select
    b.period_start,
    b.period_end,
    coalesce(c.n, 0),
    coalesce(ch.n, 0),
    coalesce(dn.n, 0),
    coalesce(del.n, 0),
    coalesce(arc.n, 0),

-- ============================================================================
-- Migration 060 (2026-09-07) — admin_stats_summary_totals() and
-- admin_stats_summary_roster(): the Sum and Averages screen's two
-- aggregation RPCs
--
-- Two functions, not one — this screen's own two layers (plan: "Grand
-- totals" and "Per-user roster") return genuinely different shapes (one
-- row cohort-wide vs. one row per account), which `returns table(...)`
-- can't mix into a single call cleanly.
--
-- Unlike the three Activity screens, this is a POINT-IN-TIME snapshot, not
-- a per-period flow — the plan's own wording is "cohort-wide, as of the
-- selected To-date." p_asof replaces the from/granularity pair the other
-- three screens take; there is no week/month bucketing here at all.
--
-- Grand totals read straight off the live requests/contacts/dialog/
-- attachments tables (created_at <= p_asof), not events -- a "how many
-- exist right now (or existed by this date, among survivors)" snapshot is
-- exactly what a live table scan gives correctly, unlike the Activity
-- screens' own per-period flow counts. The same permanent limitation
-- migration 056's own comment already gives for Contacts' Total applies
-- here too: anything hard-deleted before p_asof is invisible to a live
-- scan, so this reads as "of what survives today, how much existed by
-- then" -- not a true historical total.
--
-- Status splits (Open/Overdue/Done/Archived) in the roster reconstruct
-- Done/Archived from done_date/archived_at compared against p_asof (both
-- survive independently of anything else, so this is a real, accurate
-- reconstruction for any past date) -- but Overdue is only ever computed
-- when p_asof is today: it depends on comparing due_date/due_time against
-- the clock, and due dates can be edited after the fact, so "Overdue as
-- of last month" would silently mean "today's due dates, evaluated as if
-- it were last month," not what was actually true then. This is the exact
-- permanent limitation the plan's own "Known limitation -- Overdue"
-- section already calls out for the Activity screens; asking for a past
-- p_asof here folds would-be-Overdue rows into Open instead of guessing.
-- ============================================================================

create or replace function public.admin_stats_summary_totals(
  p_asof date,
  p_cohort text,
  p_profile_id uuid default null
)
returns table (
  user_count bigint,
  requests_total bigint,
  dialog_total bigint,
  avg_dialog_per_request numeric,
  avg_dialog_per_todo numeric,
  attachments_total_count bigint,
  attachments_total_size_kb numeric,
  avg_request_description_size numeric,
  todos_total bigint,
  avg_todo_description_size numeric
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_is_admin boolean;
begin
  select is_admin into v_is_admin from public.profiles where id = auth.uid();
  if not coalesce(v_is_admin, false) then
    raise exception 'Not authorized.';
  end if;
  if p_cohort not in ('all', 'beta', 'profile') then
    raise exception 'Invalid cohort.';
  end if;

  return query
  with cohort as (
    select u.id
    from auth.users u
    where p_cohort = 'all'
       or (p_cohort = 'profile' and u.id = p_profile_id)
       or (p_cohort = 'beta' and lower(u.email) in (select lower(email) from public.beta_allowlist))
  ),
  req as (
    select * from public.requests r
    where r.owner_id in (select id from cohort)
      and r.contact_id is not null
      and r.created_at::date <= p_asof
  ),
  td as (
    select * from public.requests r
    where r.owner_id in (select id from cohort)
      and r.contact_id is null
      and r.created_at::date <= p_asof
  ),
  dlg_req as (
    select count(*) as n from public.dialog d join req on req.id = d.request_id
      where d.created_at::date <= p_asof
  ),
  dlg_td as (
    select count(*) as n from public.dialog d join td on td.id = d.request_id
      where d.created_at::date <= p_asof
  ),
  att as (
    select count(*) as n, coalesce(sum(a.size_bytes), 0) as bytes
    from public.attachments a
    where a.request_id in (select id from req union select id from td)
      and a.created_at::date <= p_asof
  )
  select
    (select count(*) from cohort),
    (select count(*) from req),
    (select n from dlg_req) + (select n from dlg_td),
    case when (select count(*) from req) = 0 then null
      else round((select n from dlg_req)::numeric / (select count(*) from req), 2) end,
    case when (select count(*) from td) = 0 then null
      else round((select n from dlg_td)::numeric / (select count(*) from td), 2) end,
    (select n from att),
    round((select bytes from att)::numeric / 1024, 1),
    (select round(avg(length(description)), 1) from req),
    (select count(*) from td),
    (select round(avg(length(description)), 1) from td);
end;
$$;

-- admin_stats_summary_roster — one row per cohort account. See this

create or replace function public.admin_stats_summary_roster(
  p_asof date,
  p_cohort text,
  p_profile_id uuid default null
)
returns table (
  account_id uuid,
  email text,
  display_name text,
  contact_count bigint,
  requests_sent_total bigint,
  requests_sent_open bigint,
  requests_sent_overdue bigint,
  requests_sent_done bigint,
  requests_sent_archived bigint,
  requests_received_total bigint,
  requests_received_open bigint,
  requests_received_overdue bigint,
  requests_received_done bigint,
  requests_received_archived bigint,
  dialog_count bigint,
  attachments_count bigint,
  attachments_avg_size_kb numeric,
  todos_total bigint,
  todos_open bigint,
  todos_done bigint,
  todos_archived bigint
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_is_admin boolean;
  v_overdue_computable boolean;
begin
  select is_admin into v_is_admin from public.profiles where id = auth.uid();
  if not coalesce(v_is_admin, false) then
    raise exception 'Not authorized.';
  end if;
  if p_cohort not in ('all', 'beta', 'profile') then
    raise exception 'Invalid cohort.';
  end if;

  v_overdue_computable := (p_asof = current_date);

  return query
  with cohort as (
    select u.id, u.email
    from auth.users u
    where p_cohort = 'all'
       or (p_cohort = 'profile' and u.id = p_profile_id)
       or (p_cohort = 'beta' and lower(u.email) in (select lower(email) from public.beta_allowlist))
  )
  select
    u.id as account_id,
    u.email::text,
    p.display_name,

    (select count(*) from public.contacts c
      where c.owner_id = u.id and c.created_at::date <= p_asof),

    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is not null and r.created_at::date <= p_asof),
    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is not null and r.created_at::date <= p_asof
        and (r.archived_at is null or r.archived_at::date > p_asof)
        and (r.done_date is null or r.done_date > p_asof)
        and not (v_overdue_computable and r.due_date < current_date)),
    (select case when not v_overdue_computable then 0 else count(*) end from public.requests r
      where r.owner_id = u.id and r.contact_id is not null and r.created_at::date <= p_asof
        and (r.archived_at is null or r.archived_at::date > p_asof)
        and (r.done_date is null or r.done_date > p_asof)
        and v_overdue_computable and r.due_date < current_date),

create or replace function public.admin_stats_contacts(
  p_from date,
  p_to date,
  p_granularity text,
  p_cohort text,
  p_profile_id uuid default null
)
returns table (
  period_start date,
  period_end date,
  added bigint,
  deleted bigint,
  with_phone bigint,
  with_notes bigint,
  total bigint
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_is_admin boolean;
begin
  select is_admin into v_is_admin from public.profiles where id = auth.uid();
  if not coalesce(v_is_admin, false) then
    raise exception 'Not authorized.';
  end if;
  if p_granularity not in ('week', 'month') then
    raise exception 'Invalid granularity.';
  end if;
  if p_cohort not in ('all', 'beta', 'profile', 'free', 'subscriber') then
    raise exception 'Invalid cohort.';
  end if;

  return query
  with cohort as (
    select u.id
    from auth.users u
    left join public.profiles pr on pr.id = u.id
    where p_cohort = 'all'
       or (p_cohort = 'profile' and u.id = p_profile_id)
       or (p_cohort = 'beta' and lower(u.email) in (select lower(email) from public.beta_allowlist))
       or (p_cohort = 'free' and coalesce(pr.tier, 'free') = 'free')
       or (p_cohort = 'subscriber' and pr.tier = 'subscriber')
  ),
  periods as (
    select
      gs::date as period_start,
      (case when p_granularity = 'week'
         then (gs + interval '6 days')
         else (date_trunc('month', gs) + interval '1 month' - interval '1 day')
       end)::date as period_end_natural
    from generate_series(
      case when p_granularity = 'week'
        then date_trunc('week', p_from::timestamp + interval '1 day') - interval '1 day'
        else date_trunc('month', p_from::timestamp)
      end,
      p_to::timestamp,
      case when p_granularity = 'week' then interval '7 days' else interval '1 month' end
    ) as gs
  ),
  bucketed as (
    select p.period_start, least(p.period_end_natural, p_to) as period_end
    from periods p
  ),
  ev_created as (
    select
      (case when p_granularity = 'week'
         then (date_trunc('week', e.at + interval '1 day') - interval '1 day')
         else date_trunc('month', e.at)
       end)::date as period_start,
      count(*) as added,
      count(*) filter (where (e.detail->>'has_phone')::boolean is true) as with_phone,
      count(*) filter (where (e.detail->>'has_notes')::boolean is true) as with_notes
    from public.events e
    where e.subject_type = 'contact' and e.action = 'created'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  ev_deleted as (
    select
      (case when p_granularity = 'week'
         then (date_trunc('week', e.at + interval '1 day') - interval '1 day')
         else date_trunc('month', e.at)
       end)::date as period_start,
      count(*) as deleted
    from public.events e
    where e.subject_type = 'contact' and e.action = 'deleted'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  )
  select
    b.period_start,
    b.period_end,
    coalesce(c.added, 0) as added,
    coalesce(dl.deleted, 0) as deleted,
    coalesce(c.with_phone, 0) as with_phone,
    coalesce(c.with_notes, 0) as with_notes,
    (
      select count(*) from public.contacts ct
      where ct.owner_id in (select id from cohort)
        and ct.created_at::date <= b.period_end
    ) as total
  from bucketed b
  left join ev_created c on c.period_start = b.period_start
  left join ev_deleted dl on dl.period_start = b.period_start
  order by b.period_start;
end;
$$;

grant execute on function public.admin_stats_contacts(date, date, text, text, uuid) to authenticated;

       or (p_cohort = 'profile' and u.id = p_profile_id)
       or (p_cohort = 'beta' and lower(u.email) in (select lower(email) from public.beta_allowlist))

    select
      gs::date as period_start,
      (case when p_granularity = 'week'
         then (gs + interval '6 days')
         else (date_trunc('month', gs) + interval '1 month' - interval '1 day')

    join auth.users u on lower(u.email) = lower(trim(c.email))
    where e.subject_type = 'request' and e.action = 'changed' and e.actor_user is null
      and u.id in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  sent_done_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'request' and e.action = 'done'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  received_done_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    join public.requests r on r.id = e.request_id
    join public.contacts c on c.id = r.contact_id
    join auth.users u on lower(u.email) = lower(trim(c.email))
    where e.subject_type = 'request' and e.action = 'done' and e.actor_user is null
      and u.id in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  deleted_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'request' and e.action = 'deleted'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  archived_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'request' and e.action = 'archived'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  unarchived_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'request' and e.action = 'unarchived'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  sent_attachments_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'attachment' and e.action = 'created'
      and (e.detail->>'side') = 'sent'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  received_attachments_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    join public.requests r on r.id = e.request_id
    join public.contacts c on c.id = r.contact_id
    join auth.users u on lower(u.email) = lower(trim(c.email))
    where e.subject_type = 'attachment' and e.action = 'created'
      and (e.detail->>'side') = 'received'
      and u.id in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  sent_dialog_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', d.created_at + interval '1 day') - interval '1 day') else date_trunc('month', d.created_at) end)::date as period_start,
      count(*) as n
    from public.dialog d
    join public.requests r on r.id = d.request_id
    where r.contact_id is not null
      and d.author_user_id = r.owner_id
      and r.owner_id in (select id from cohort)
      and d.created_at::date between p_from and p_to
    group by 1
  ),
  received_dialog_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', d.created_at + interval '1 day') - interval '1 day') else date_trunc('month', d.created_at) end)::date as period_start,
      count(*) as n
    from public.dialog d
    join public.requests r on r.id = d.request_id
    join public.contacts c on c.id = r.contact_id
    join auth.users u on lower(u.email) = lower(trim(c.email))
    where r.contact_id is not null
      and d.author_user_id is distinct from r.owner_id
      and u.id in (select id from cohort)
      and d.created_at::date between p_from and p_to
    group by 1
  )
  select
    b.period_start,
    b.period_end,
    coalesce(sc.n, 0), coalesce(rc.n, 0),
    coalesce(sch.n, 0), coalesce(rch.n, 0),
    coalesce(sd.n, 0), coalesce(rd.n, 0),
    coalesce(del.n, 0),
    coalesce(arc.n, 0),
    coalesce(unarc.n, 0),
    coalesce(sa.n, 0), coalesce(ra.n, 0),
    coalesce(sdlg.n, 0), coalesce(rdlg.n, 0)
  from bucketed b
  left join sent_created_t sc on sc.period_start = b.period_start
  left join received_created_t rc on rc.period_start = b.period_start
  left join sent_changed_t sch on sch.period_start = b.period_start
  left join received_changed_t rch on rch.period_start = b.period_start
  left join sent_done_t sd on sd.period_start = b.period_start
  left join received_done_t rd on rd.period_start = b.period_start
  left join deleted_t del on del.period_start = b.period_start
  left join archived_t arc on arc.period_start = b.period_start
  left join unarchived_t unarc on unarc.period_start = b.period_start
  left join sent_attachments_t sa on sa.period_start = b.period_start
  left join received_attachments_t ra on ra.period_start = b.period_start
  left join sent_dialog_t sdlg on sdlg.period_start = b.period_start
  left join received_dialog_t rdlg on rdlg.period_start = b.period_start
  order by b.period_start;
end;
$$;

create or replace function public.admin_stats_todos(
  p_from date,
  p_to date,
  p_granularity text,
  p_cohort text,
  p_profile_id uuid default null
)
returns table (
  period_start date,
  period_end date,
  created bigint,
  changed bigint,
  done bigint,
  deleted bigint,
  archived bigint,
  unarchived bigint,
  attachments bigint,
  dialog bigint
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_is_admin boolean;
begin
  select is_admin into v_is_admin from public.profiles where id = auth.uid();
  if not coalesce(v_is_admin, false) then
    raise exception 'Not authorized.';
  end if;
  if p_granularity not in ('week', 'month') then
    raise exception 'Invalid granularity.';
  end if;
  if p_cohort not in ('all', 'beta', 'profile', 'free', 'subscriber') then
    raise exception 'Invalid cohort.';
  end if;

  return query
  with cohort as (
    select u.id
    from auth.users u
    left join public.profiles pr on pr.id = u.id
    where p_cohort = 'all'
       or (p_cohort = 'profile' and u.id = p_profile_id)
       or (p_cohort = 'beta' and lower(u.email) in (select lower(email) from public.beta_allowlist))
       or (p_cohort = 'free' and coalesce(pr.tier, 'free') = 'free')
       or (p_cohort = 'subscriber' and pr.tier = 'subscriber')
  ),
  periods as (
    select
      gs::date as period_start,
      (case when p_granularity = 'week'
         then (gs + interval '6 days')
         else (date_trunc('month', gs) + interval '1 month' - interval '1 day')
       end)::date as period_end_natural
    from generate_series(
      case when p_granularity = 'week'
        then date_trunc('week', p_from::timestamp + interval '1 day') - interval '1 day'
        else date_trunc('month', p_from::timestamp)
      end,
      p_to::timestamp,
      case when p_granularity = 'week' then interval '7 days' else interval '1 month' end
    ) as gs
  ),
  bucketed as (
    select p.period_start, least(p.period_end_natural, p_to) as period_end
    from periods p
  ),
  created_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'todo' and e.action = 'created'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  changed_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'todo' and e.action = 'changed'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  done_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'todo' and e.action = 'done'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  deleted_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'todo' and e.action = 'deleted'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  archived_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'todo' and e.action = 'archived'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  unarchived_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    where e.subject_type = 'todo' and e.action = 'unarchived'
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  attachments_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', e.at + interval '1 day') - interval '1 day') else date_trunc('month', e.at) end)::date as period_start,
      count(*) as n
    from public.events e
    join public.requests r on r.id = e.request_id
    where e.subject_type = 'attachment' and e.action = 'created'
      and r.contact_id is null
      and e.actor_user in (select id from cohort)
      and e.at::date between p_from and p_to
    group by 1
  ),
  dialog_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', d.created_at + interval '1 day') - interval '1 day') else date_trunc('month', d.created_at) end)::date as period_start,
      count(*) as n
    from public.dialog d
    join public.requests r on r.id = d.request_id
    where r.contact_id is null
      and r.owner_id in (select id from cohort)
      and d.created_at::date between p_from and p_to
    group by 1
  )
  select
    b.period_start,
    b.period_end,
    coalesce(c.n, 0),
    coalesce(ch.n, 0),
    coalesce(dn.n, 0),
    coalesce(del.n, 0),
    coalesce(arc.n, 0),
    coalesce(unarc.n, 0),
    coalesce(att.n, 0),
    coalesce(dlg.n, 0)
  from bucketed b
  left join created_t c on c.period_start = b.period_start
  left join changed_t ch on ch.period_start = b.period_start
  left join done_t dn on dn.period_start = b.period_start
  left join deleted_t del on del.period_start = b.period_start
  left join archived_t arc on arc.period_start = b.period_start
  left join unarchived_t unarc on unarc.period_start = b.period_start
  left join attachments_t att on att.period_start = b.period_start
  left join dialog_t dlg on dlg.period_start = b.period_start
  order by b.period_start;
end;
$$;

revoke all on function public.admin_stats_todos(date, date, text, text, uuid) from public;
grant execute on function public.admin_stats_todos(date, date, text, text, uuid) to authenticated;

create or replace function public.admin_stats_summary_totals(
  p_asof date,
  p_cohort text,
  p_profile_id uuid default null
)
returns table (
  user_count bigint,
  requests_total bigint,
  dialog_total bigint,
  avg_dialog_per_request numeric,
  avg_dialog_per_todo numeric,
  attachments_total_count bigint,
  attachments_total_size_kb numeric,
  avg_request_description_size numeric,
  todos_total bigint,
  avg_todo_description_size numeric
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_is_admin boolean;
begin
  select is_admin into v_is_admin from public.profiles where id = auth.uid();
  if not coalesce(v_is_admin, false) then
    raise exception 'Not authorized.';
  end if;
  if p_cohort not in ('all', 'beta', 'profile', 'free', 'subscriber') then
    raise exception 'Invalid cohort.';
  end if;

  return query
  with cohort as (
    select u.id
    from auth.users u
    left join public.profiles pr on pr.id = u.id
    where p_cohort = 'all'
       or (p_cohort = 'profile' and u.id = p_profile_id)
       or (p_cohort = 'beta' and lower(u.email) in (select lower(email) from public.beta_allowlist))
       or (p_cohort = 'free' and coalesce(pr.tier, 'free') = 'free')
       or (p_cohort = 'subscriber' and pr.tier = 'subscriber')
  ),
  req as (
    select * from public.requests r
    where r.owner_id in (select id from cohort)
      and r.contact_id is not null
      and r.created_at::date <= p_asof
  ),
  td as (
    select * from public.requests r
    where r.owner_id in (select id from cohort)
      and r.contact_id is null
      and r.created_at::date <= p_asof
  ),
  dlg_req as (
    select count(*) as n from public.dialog d join req on req.id = d.request_id
      where d.created_at::date <= p_asof
  ),
  dlg_td as (
    select count(*) as n from public.dialog d join td on td.id = d.request_id
      where d.created_at::date <= p_asof
  ),
  att as (
    select count(*) as n, coalesce(sum(a.size_bytes), 0) as bytes
    from public.attachments a
    where a.request_id in (select id from req union select id from td)
      and a.created_at::date <= p_asof
  )
  select
    (select count(*) from cohort),
    (select count(*) from req),
    (select n from dlg_req) + (select n from dlg_td),
    case when (select count(*) from req) = 0 then null
      else round((select n from dlg_req)::numeric / (select count(*) from req), 2) end,
    case when (select count(*) from td) = 0 then null
      else round((select n from dlg_td)::numeric / (select count(*) from td), 2) end,
    (select n from att),
    round((select bytes from att)::numeric / 1024, 1),
    (select round(avg(length(description)), 1) from req),
    (select count(*) from td),
    (select round(avg(length(description)), 1) from td);
end;
$$;

revoke all on function public.admin_stats_summary_totals(date, text, uuid) from public;
grant execute on function public.admin_stats_summary_totals(date, text, uuid) to authenticated;

create or replace function public.admin_stats_summary_roster(
  p_asof date,
  p_cohort text,
  p_profile_id uuid default null
)
returns table (
  account_id uuid,
  email text,
  display_name text,
  contact_count bigint,
  requests_sent_total bigint,
  requests_sent_open bigint,
  requests_sent_overdue bigint,
  requests_sent_done bigint,
  requests_sent_archived bigint,
  requests_received_total bigint,
  requests_received_open bigint,
  requests_received_overdue bigint,
  requests_received_done bigint,
  requests_received_archived bigint,
  dialog_count bigint,
  attachments_count bigint,
  attachments_avg_size_kb numeric,
  todos_total bigint,
  todos_open bigint,
  todos_done bigint,
  todos_archived bigint
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_is_admin boolean;
  v_overdue_computable boolean;
begin
  select is_admin into v_is_admin from public.profiles where id = auth.uid();
  if not coalesce(v_is_admin, false) then
    raise exception 'Not authorized.';
  end if;
  if p_cohort not in ('all', 'beta', 'profile', 'free', 'subscriber') then
    raise exception 'Invalid cohort.';
  end if;

  v_overdue_computable := (p_asof = current_date);

  return query
  with cohort as (
    select u.id, u.email
    from auth.users u
    left join public.profiles pr on pr.id = u.id
    where p_cohort = 'all'
       or (p_cohort = 'profile' and u.id = p_profile_id)
       or (p_cohort = 'beta' and lower(u.email) in (select lower(ba.email) from public.beta_allowlist ba))
       or (p_cohort = 'free' and coalesce(pr.tier, 'free') = 'free')
       or (p_cohort = 'subscriber' and pr.tier = 'subscriber')
  )
  select
    u.id as account_id,
    u.email::text,
    p.display_name,

    (select count(*) from public.contacts c
      where c.owner_id = u.id and c.created_at::date <= p_asof),

    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is not null and r.created_at::date <= p_asof),
    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is not null and r.created_at::date <= p_asof
        and (r.archived_at is null or r.archived_at::date > p_asof)
        and (r.done_date is null or r.done_date > p_asof)
        and not (v_overdue_computable and r.due_date < current_date)),
    (select case when not v_overdue_computable then 0 else count(*) end from public.requests r
      where r.owner_id = u.id and r.contact_id is not null and r.created_at::date <= p_asof
        and (r.archived_at is null or r.archived_at::date > p_asof)
        and (r.done_date is null or r.done_date > p_asof)
        and v_overdue_computable and r.due_date < current_date),
    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is not null and r.created_at::date <= p_asof
        and r.done_date is not null and r.done_date <= p_asof
        and (r.archived_at is null or r.archived_at::date > p_asof)),
    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is not null and r.created_at::date <= p_asof
        and r.archived_at is not null and r.archived_at::date <= p_asof),

    (select count(*) from public.requests r
      join public.contacts c on c.id = r.contact_id
      where lower(trim(c.email)) = lower(u.email) and r.created_at::date <= p_asof),
    (select count(*) from public.requests r
      join public.contacts c on c.id = r.contact_id
      where lower(trim(c.email)) = lower(u.email) and r.created_at::date <= p_asof
        and (r.archived_at is null or r.archived_at::date > p_asof)
        and (r.done_date is null or r.done_date > p_asof)
        and not (v_overdue_computable and r.due_date < current_date)),
    (select case when not v_overdue_computable then 0 else count(*) end from public.requests r
      join public.contacts c on c.id = r.contact_id
      where lower(trim(c.email)) = lower(u.email) and r.created_at::date <= p_asof
        and (r.archived_at is null or r.archived_at::date > p_asof)
        and (r.done_date is null or r.done_date > p_asof)
        and v_overdue_computable and r.due_date < current_date),
    (select count(*) from public.requests r
      join public.contacts c on c.id = r.contact_id
      where lower(trim(c.email)) = lower(u.email) and r.created_at::date <= p_asof
        and r.done_date is not null and r.done_date <= p_asof
        and (r.archived_at is null or r.archived_at::date > p_asof)),
    (select count(*) from public.requests r
      join public.contacts c on c.id = r.contact_id
      where lower(trim(c.email)) = lower(u.email) and r.created_at::date <= p_asof
        and r.archived_at is not null and r.archived_at::date <= p_asof),

    (select count(*) from public.dialog d
      join public.requests r on r.id = d.request_id
      where r.owner_id = u.id and d.created_at::date <= p_asof),

    (select count(*) from public.attachments a
      join public.requests r on r.id = a.request_id
      where r.owner_id = u.id and a.created_at::date <= p_asof),
    (select case when count(*) = 0 then null else round(sum(a.size_bytes)::numeric / count(*) / 1024, 1) end
      from public.attachments a join public.requests r on r.id = a.request_id
      where r.owner_id = u.id and a.created_at::date <= p_asof),

    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is null and r.created_at::date <= p_asof),
    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is null and r.created_at::date <= p_asof
        and (r.archived_at is null or r.archived_at::date > p_asof)
        and (r.done_date is null or r.done_date > p_asof)),
    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is null and r.created_at::date <= p_asof
        and r.done_date is not null and r.done_date <= p_asof
        and (r.archived_at is null or r.archived_at::date > p_asof)),
    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is null and r.created_at::date <= p_asof
        and r.archived_at is not null and r.archived_at::date <= p_asof)

  from cohort u
  left join public.profiles p on p.id = u.id
  order by u.email;
end;
$$;

revoke all on function public.admin_stats_summary_roster(date, text, uuid) from public;

-- Migration 064 (2026-09-07) -- admin_stats_accounts(): new "Accounts --
-- Activity" screen, a 5th admin stats screen added same-session
-- (Jim: "there should be another activity screen 'Accounts' to show
-- by-period new free and new subscribed accounts and a total new
-- accounts")
--
-- Unlike Contacts/Requests/ToDos, this reads straight off auth.users --
-- no events dependency at all. Those three screens read from events
-- specifically because Contacts/Requests/ToDos/Attachments are hard-
-- deleted, so a live table scan silently undercounts anything removed
-- since. Accounts have no delete path anywhere in this app, so there is
-- no undercount risk to guard against -- a live scan of auth.users,
-- bucketed by created_at, is fully accurate on its own.
--
-- Cohort is deliberately narrower than the other four admin RPCs: only
-- 'all' and 'beta' -- 'profile' is degenerate here (a single account's
-- own "new accounts per period" is always one row of 1, everywhere else
-- 0) and 'free'/'subscriber' would be redundant, since New Free vs. New
-- Subscribed is exactly the split this screen's own two columns already
-- show. p_profile_id is kept as an accepted-but-ignored parameter purely
-- so /api/admin/stats/export's generic ENTITY_CONFIG path (which always
-- forwards p_profile_id to whichever RPC it's calling) doesn't need a
-- special case for this one screen the way entity=summary already has.
--
-- Same known limitation as the Free/Subscribed cohort filter (migration
-- 063): profiles.tier has no history, so "New Subscribed in July" means
-- "accounts created in July that are subscribers TODAY," not tier-at-
-- signup. Documented, not fixable without a tier-history table this app
-- doesn't have.
-- ============================================================================

create or replace function public.admin_stats_accounts(
  p_from date,
  p_to date,
  p_granularity text,
  p_cohort text default 'all',
  p_profile_id uuid default null
)
returns table (
  period_start date,
  period_end date,
  new_free bigint,
  new_subscribed bigint,
  total bigint
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_is_admin boolean;
begin
  select pr.is_admin into v_is_admin from public.profiles pr where pr.id = auth.uid();
  if not coalesce(v_is_admin, false) then
    raise exception 'Not authorized.';
  end if;
  if p_granularity not in ('week', 'month') then
    raise exception 'Invalid granularity.';
  end if;
  if p_cohort not in ('all', 'beta') then
    raise exception 'Invalid cohort.';
  end if;

  return query
  with cohort as (
    select u.id, u.created_at, coalesce(pr.tier, 'free') as tier
    from auth.users u
    left join public.profiles pr on pr.id = u.id
    where p_cohort = 'all'
       or (p_cohort = 'beta' and lower(u.email) in (select lower(ba.email) from public.beta_allowlist ba))
  ),
  periods as (
    select
      gs::date as period_start,
      (case when p_granularity = 'week'
         then (gs + interval '6 days')
         else (date_trunc('month', gs) + interval '1 month' - interval '1 day')
       end)::date as period_end_natural
    from generate_series(
      case when p_granularity = 'week'
        then date_trunc('week', p_from::timestamp + interval '1 day') - interval '1 day'
        else date_trunc('month', p_from::timestamp)
      end,
      p_to::timestamp,
      case when p_granularity = 'week' then interval '7 days' else interval '1 month' end
    ) as gs
  ),
  bucketed as (
    select p.period_start, least(p.period_end_natural, p_to) as period_end
    from periods p
  ),
  new_free_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', c.created_at + interval '1 day') - interval '1 day') else date_trunc('month', c.created_at) end)::date as period_start,
      count(*) as n
    from cohort c
    where c.tier = 'free'
      and c.created_at::date between p_from and p_to
    group by 1
  ),
  new_subscribed_t as (
    select
      (case when p_granularity = 'week' then (date_trunc('week', c.created_at + interval '1 day') - interval '1 day') else date_trunc('month', c.created_at) end)::date as period_start,
      count(*) as n
    from cohort c
    where c.tier = 'subscriber'
      and c.created_at::date between p_from and p_to
    group by 1
  )
  select
    b.period_start,
    b.period_end,
    coalesce(nf.n, 0),
    coalesce(ns.n, 0),
    coalesce(nf.n, 0) + coalesce(ns.n, 0)
  from bucketed b
  left join new_free_t nf on nf.period_start = b.period_start
  left join new_subscribed_t ns on ns.period_start = b.period_start
  order by b.period_start;
end;
$$;

revoke all on function public.admin_stats_accounts(date, date, text, text, uuid) from public;
grant execute on function public.admin_stats_accounts(date, date, text, text, uuid) to authenticated;

-- Migration 065 (2026-09-07) -- Sum and Averages reformat: replaces the
-- Grand-Totals stat-tile block with an Entities table (Accounts, Contacts,
-- Req's Sent, Req's Received, ToDos -- Count Total, Avg Descr char's, and
-- an Open/Overdue/Done/Archived breakdown) and a new Volume table
-- (Dialogs, Attachments -- Count Total and Avg Descr/Size), and simplifies
-- the per-account Roster to Account/Contacts/Req Sent/Req Rec'd/ToDos/
-- Dialogs/Atch's/Avg Atch Size. Per
-- docs/WYP_Admin_Statistics_Specification_v1_0.docx (owner-approved
-- design, 2026-09-07). Percentages and the Volume table's four "average
-- count per" ratios are computed client-side from the raw counts this
-- migration returns (AdminSummaryStatsForm.tsx: pct()/ratio()), not baked
-- into the RPC, so the on-screen table, the print view, and the .xlsx
-- export share one rounding rule instead of three.
--
-- admin_stats_summary_totals() gains full Open/Overdue/Done/Archived raw
-- counts for Req's Sent, Req's Received, AND ToDos. ToDos never had any
-- of this in the totals RPC before, and the roster's own todos_* columns
-- never had an Overdue count either -- even though the main app's ToDos
-- filter chips have supported a real Overdue state since 2026-08-12
-- (ToDos gain an Overdue chip). ToDos Overdue here follows the exact same
-- "not archived, not done, due_date < current_date, and only ever
-- nonzero when p_asof = current_date" rule already used for Requests --
-- this closes a real gap between this screen and actual app behavior,
-- not just a cosmetic addition. (The PRD/UI spec text that still says
-- "Overdue does not apply to ToDos" is a separate, already-flagged
-- documentation correction -- not addressed by this migration.)
--
-- admin_stats_summary_roster() drops display_name (Account is shown as
-- email only, per owner decision) and the entire per-account Open/
-- Overdue/Done/Archived breakdown -- that detail remains available by
-- switching the existing Accounts filter to a specific account, which
-- re-scopes admin_stats_summary_totals() itself to that one account
-- instead of duplicating the breakdown in every roster row.
--
-- Dialogs/Attachments totals keep the pre-existing scope (entries on
-- Requests/ToDos OWNED by the cohort -- not entries on requests the
-- cohort merely received), matching migration 060's own original
-- semantics; this migration does not expand that scope.
--
-- Postgres refuses `create or replace function` when the OUT-parameter
-- row type changes (42P13: "cannot change return type of existing
-- function") -- both functions below change shape from their migration
-- 063 versions, so each needs an explicit drop first. Added after the
-- first run attempt hit exactly this error (owner action, 2026-09-07).
-- ============================================================================

drop function if exists public.admin_stats_summary_totals(date, text, uuid);

create or replace function public.admin_stats_summary_totals(
  p_asof date,
  p_cohort text,
  p_profile_id uuid default null
)
returns table (
  accounts_count bigint,
  contacts_count bigint,

  requests_sent_total bigint,
  requests_sent_open bigint,
  requests_sent_overdue bigint,
  requests_sent_done bigint,
  requests_sent_archived bigint,
  requests_sent_avg_description numeric,

  requests_received_total bigint,
  requests_received_open bigint,
  requests_received_overdue bigint,
  requests_received_done bigint,
  requests_received_archived bigint,
  requests_received_avg_description numeric,

  todos_total bigint,
  todos_open bigint,
  todos_overdue bigint,
  todos_done bigint,
  todos_archived bigint,
  todos_avg_description numeric,

  dialog_total bigint,
  dialog_avg_description numeric,

  attachments_total bigint,
  attachments_avg_size_kb numeric
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_is_admin boolean;
  v_overdue_computable boolean;
begin
  select is_admin into v_is_admin from public.profiles where id = auth.uid();
  if not coalesce(v_is_admin, false) then
    raise exception 'Not authorized.';
  end if;
  if p_cohort not in ('all', 'beta', 'profile', 'free', 'subscriber') then
    raise exception 'Invalid cohort.';
  end if;

  v_overdue_computable := (p_asof = current_date);

  return query
  with cohort as (
    select u.id, u.email
    from auth.users u
    left join public.profiles pr on pr.id = u.id
    where p_cohort = 'all'
       or (p_cohort = 'profile' and u.id = p_profile_id)
       or (p_cohort = 'beta' and lower(u.email) in (select lower(ba.email) from public.beta_allowlist ba))
       or (p_cohort = 'free' and coalesce(pr.tier, 'free') = 'free')
       or (p_cohort = 'subscriber' and pr.tier = 'subscriber')
  ),
  sent as (
    select r.* from public.requests r
    where r.owner_id in (select id from cohort)
      and r.contact_id is not null
      and r.created_at::date <= p_asof
  ),
  received as (
    select r.* from public.requests r
    join public.contacts c on c.id = r.contact_id
    where lower(trim(c.email)) in (select lower(email) from cohort)
      and r.created_at::date <= p_asof
  ),
  td as (
    select r.* from public.requests r
    where r.owner_id in (select id from cohort)
      and r.contact_id is null
      and r.created_at::date <= p_asof
  ),
  dlg as (
    select count(*) as n, coalesce(round(avg(length(d.body))), 0) as avg_len
    from public.dialog d
    where d.created_at::date <= p_asof
      and d.request_id in (select id from sent union select id from td)
  ),
  att as (
    select count(*) as n, coalesce(sum(a.size_bytes), 0) as bytes
    from public.attachments a
    where a.created_at::date <= p_asof
      and a.request_id in (select id from sent union select id from td)
  )
  select
    (select count(*) from cohort),
    (select count(*) from public.contacts c
      where c.owner_id in (select id from cohort) and c.created_at::date <= p_asof),

    (select count(*) from sent),
    (select count(*) from sent where (archived_at is null or archived_at::date > p_asof)
       and (done_date is null or done_date > p_asof)
       and not (v_overdue_computable and due_date < current_date)),
    (select case when not v_overdue_computable then 0 else count(*) end from sent
       where (archived_at is null or archived_at::date > p_asof)
       and (done_date is null or done_date > p_asof)
       and v_overdue_computable and due_date < current_date),
    (select count(*) from sent where done_date is not null and done_date <= p_asof
       and (archived_at is null or archived_at::date > p_asof)),
    (select count(*) from sent where archived_at is not null and archived_at::date <= p_asof),
    (select round(avg(length(description)), 0) from sent),

    (select count(*) from received),
    (select count(*) from received where (archived_at is null or archived_at::date > p_asof)
       and (done_date is null or done_date > p_asof)
       and not (v_overdue_computable and due_date < current_date)),
    (select case when not v_overdue_computable then 0 else count(*) end from received
       where (archived_at is null or archived_at::date > p_asof)
       and (done_date is null or done_date > p_asof)
       and v_overdue_computable and due_date < current_date),
    (select count(*) from received where done_date is not null and done_date <= p_asof
       and (archived_at is null or archived_at::date > p_asof)),
    (select count(*) from received where archived_at is not null and archived_at::date <= p_asof),
    (select round(avg(length(description)), 0) from received),

    (select count(*) from td),
    (select count(*) from td where (archived_at is null or archived_at::date > p_asof)
       and (done_date is null or done_date > p_asof)
       and not (v_overdue_computable and due_date < current_date)),
    (select case when not v_overdue_computable then 0 else count(*) end from td
       where (archived_at is null or archived_at::date > p_asof)
       and (done_date is null or done_date > p_asof)
       and v_overdue_computable and due_date < current_date),
    (select count(*) from td where done_date is not null and done_date <= p_asof
       and (archived_at is null or archived_at::date > p_asof)),
    (select count(*) from td where archived_at is not null and archived_at::date <= p_asof),
    (select round(avg(length(description)), 0) from td),

    (select n from dlg),
    (select avg_len from dlg),

    (select n from att),
    (select case when (select n from att) = 0 then null
       else round((select bytes from att)::numeric / (select n from att) / 1024, 0) end);
end;
$$;

revoke all on function public.admin_stats_summary_totals(date, text, uuid) from public;
grant execute on function public.admin_stats_summary_totals(date, text, uuid) to authenticated;

create or replace function public.admin_stats_summary_roster(
  p_asof date,
  p_cohort text,
  p_profile_id uuid default null
)
returns table (
  account_id uuid,
  email text,
  contact_count bigint,
  requests_sent_total bigint,
  requests_received_total bigint,
  todos_total bigint,
  dialog_count bigint,
  attachments_count bigint,
  attachments_avg_size_kb numeric
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_is_admin boolean;
begin
  select is_admin into v_is_admin from public.profiles where id = auth.uid();
  if not coalesce(v_is_admin, false) then
    raise exception 'Not authorized.';
  end if;
  if p_cohort not in ('all', 'beta', 'profile', 'free', 'subscriber') then
    raise exception 'Invalid cohort.';
  end if;

  return query
  with cohort as (
    select u.id, u.email
    from auth.users u
    left join public.profiles pr on pr.id = u.id
    where p_cohort = 'all'
       or (p_cohort = 'profile' and u.id = p_profile_id)
       or (p_cohort = 'beta' and lower(u.email) in (select lower(ba.email) from public.beta_allowlist ba))
       or (p_cohort = 'free' and coalesce(pr.tier, 'free') = 'free')
       or (p_cohort = 'subscriber' and pr.tier = 'subscriber')
  )
  select
    u.id as account_id,
    u.email::text,

    (select count(*) from public.contacts c
      where c.owner_id = u.id and c.created_at::date <= p_asof),

    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is not null and r.created_at::date <= p_asof),

    (select count(*) from public.requests r
      join public.contacts c on c.id = r.contact_id
      where lower(trim(c.email)) = lower(u.email) and r.created_at::date <= p_asof),

    (select count(*) from public.requests r
      where r.owner_id = u.id and r.contact_id is null and r.created_at::date <= p_asof),

    (select count(*) from public.dialog d
      join public.requests r on r.id = d.request_id
      where r.owner_id = u.id and d.created_at::date <= p_asof),

    (select count(*) from public.attachments a
      join public.requests r on r.id = a.request_id
      where r.owner_id = u.id and a.created_at::date <= p_asof),
    (select case when count(*) = 0 then null else round(sum(a.size_bytes)::numeric / count(*) / 1024, 0) end
      from public.attachments a join public.requests r on r.id = a.request_id
      where r.owner_id = u.id and a.created_at::date <= p_asof)

  from cohort u
  order by u.email;
end;
$$;

revoke all on function public.admin_stats_summary_roster(date, text, uuid) from public;

-- Migration 068 (2026-09-23) -- requests.repeat_series_id: lets the
-- "Repeating" chip (Main Screen, 2026-09-18) show the current, active
-- occurrence of a series instead of requiring occurrence_index = 1
-- specifically.
--
-- Owner-reported: a repeat series' original occurrence (occurrence_index 1)
-- can lose its own repeat_rule over time -- either because it was explicitly
-- removed on that specific row, or simply because Phase E's own generation
-- never clears repeat_rule from any row it processes (only ever sets
-- repeat_next_generated_at) -- so occurrence 1 staying the sole anchor was
-- fragile. repeat_series_id is a plain nullable uuid, set once when a repeat
-- is first added to a brand-new occurrence and copied forward onto every
-- occurrence cron Phase E generates from it (app/api/cron/tick/route.ts) --
-- the chip's own filter (MainScreen.tsx) now groups rows by this id and
-- shows, per group, whichever non-archived row currently has the highest
-- repeat_occurrence_index and still carries an active repeat_rule -- "the
-- most recent item in the series," per the owner's own framing, since
-- that's also the row future generation and any Repeat edits should
-- actually be made on.
--
-- No backfill for existing rows -- this is private-testing data only, and a
-- reliable backfill would need a fragile owner+description+due-date-
-- proximity heuristic with no real guarantee of correctness. Existing
-- repeat chains simply read as repeat_series_id = null, which the chip's
-- own grouping already treats as "this row is its own one-row series"
-- (falls back to the row's own id) -- an accepted, flagged limitation for
-- pre-existing test data; every new repeat created going forward gets a
-- real shared id.
-- ============================================================================

alter table public.requests
  add column if not exists repeat_series_id uuid;

comment on column public.requests.repeat_series_id is
  'Shared across every occurrence of one repeat series, set once when Repeat is first added to a brand-new occurrence and copied forward by cron Phase E on each generated successor. Null for a non-repeating row, and for any repeat series created before migration 068 (no backfill -- see this migration''s own header comment).';

-- round trip, and no requirement that the recipient also click Send (the
-- owner's own explicit requirement: "the user should not need to
-- additionally click the Send button"). receipt_confirmed_at is the one
-- new timestamp; there is no separate "confirmed by" column since this is
-- the same anonymous-or-signed-in recipient identity every other
-- token-scoped write in this app already treats as one actor.
--
-- Two follow-up product questions (raised in chat, not in the owner's own
-- spec) were resolved before this was built: Response Detail (the
-- signed-in recipient's own view of the same Request, never reached via
-- the emailed link) gets a read-only status line, not its own working
-- Confirm button -- the confirming action stays exclusive to the emailed
-- /r/[token] link, matching the spec exactly. Request Detail (the owner's
-- own screen) also gets a read-only status line, even though the owner's
-- spec only ever drew the checkbox on Create Request -- so the owner isn't
-- limited to the Sent list's small icon to check status.
--
-- get_request_by_token needs a real signature change (a new parameter),
-- which create-or-replace cannot apply to an existing function -- same
-- constraint this file has already hit for RETURNS TABLE functions
-- (migrations 017/021/027/040's own headers), but here it's the INPUT
-- arity that changes, not the OUTPUT shape. Dropped and recreated, same as
-- those precedents. get_received_request needs no signature change (same
-- jsonb-returning, no new parameter) and gets a plain create-or-replace.
--
-- No column-level grant needed on requests.receipt_confirmation_requested/
-- receipt_confirmed_at -- same reasoning as every other plain requests
-- column addition in this file: that table uses a blanket table-level
-- grant to authenticated, unlike profiles' own per-column convention.
-- offer_receipt_confirmation lives on profiles, so it does need one,
-- matching every other Account Options toggle added there
-- (request_reminders_enabled, todo_reminders_enabled, etc.).
-- ============================================================================

create function public.get_request_by_token(p_token text, p_confirm_receipt boolean default false)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.reminder_enabled, r.reminder_sent_at, r.overdue_reminder_enabled, r.reminder_day_of_enabled,
         r.reminder_day_of_sent_at, r.repeat_rule,
         r.receipt_confirmation_requested, r.receipt_confirmed_at,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         p.request_reminders_enabled as owner_request_reminders_enabled,
         ct.display_name as contact_name,
         r.link_expires_at, r.link_revoked_at
    into v_request
    from public.requests r
    left join public.profiles p on p.id = r.owner_id
    left join public.contacts ct on ct.id = r.contact_id
    where r.link_token_hash = v_hash;

  if v_request.id is null
     or v_request.link_revoked_at is not null
     or v_request.link_expires_at < now()
  then
    raise exception 'This link is no longer active. The sender has removed the Request it pointed to.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', v_request.id, v_request.id, 'viewed', '{}'::jsonb);

  -- Receipt Confirmation side effect (migration 069) -- fires on the same
  -- read call the "Click to confirm receipt" email link lands on, so the
  -- recipient never has to also click Send. Idempotent: a no-op once
  -- receipt_confirmed_at is already set, and a no-op if this particular
  -- Request never had the option requested in the first place.
  if p_confirm_receipt
     and v_request.receipt_confirmation_requested
     and v_request.receipt_confirmed_at is null
  then
    update public.requests set receipt_confirmed_at = now() where id = v_request.id;
    v_request.receipt_confirmed_at := now();
  end if;

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'reminder_enabled', v_request.reminder_enabled,
    'reminder_sent_at', v_request.reminder_sent_at,
    'overdue_reminder_enabled', v_request.overdue_reminder_enabled,
    'reminder_day_of_enabled', v_request.reminder_day_of_enabled,
    'reminder_day_of_sent_at', v_request.reminder_day_of_sent_at,
    'repeat_rule', v_request.repeat_rule,
    'receipt_confirmation_requested', v_request.receipt_confirmation_requested,
    'receipt_confirmed_at', v_request.receipt_confirmed_at,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'owner_request_reminders_enabled', coalesce(v_request.owner_request_reminders_enabled, false),
    'contact_name', v_request.contact_name,
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_request_by_token(text, boolean) from public;
grant execute on function public.get_request_by_token(text, boolean) to anon, authenticated;

drop function if exists public.set_response_done_as_recipient(uuid, date, time, boolean, boolean, boolean);

create function public.set_response_done_as_recipient(
  p_request_id uuid,
  p_done_date date,
  p_done_time time,
  p_reminder_enabled boolean default null,
  p_overdue_reminder_enabled boolean default null,
  p_reminder_day_of_enabled boolean default null,
  p_confirm_receipt boolean default false
)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_contact_email text;
  v_receipt_requested boolean;
  v_receipt_confirmed_at timestamptz;
begin
  select lower(trim(c.email)), r.receipt_confirmation_requested, r.receipt_confirmed_at
    into v_contact_email, v_receipt_requested, v_receipt_confirmed_at
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = p_request_id;

  if v_email = '' or v_contact_email is null or v_contact_email <> v_email then
    raise exception 'Request not found.';
  end if;

  update public.requests
    set done_date = p_done_date,
        done_time = p_done_time,
        received_archived_at = case when p_done_date is null then received_archived_at else null end,
        reminder_enabled = coalesce(p_reminder_enabled, reminder_enabled),
        overdue_reminder_enabled = coalesce(p_overdue_reminder_enabled, overdue_reminder_enabled),
        reminder_day_of_enabled = coalesce(p_reminder_day_of_enabled, reminder_day_of_enabled),
        -- Receipt Confirmation (migration 070) -- only ever moves null -> now(),
        -- same idempotent, one-way behavior as get_request_by_token's own
        -- ?confirm=1 side effect (migration 069).
        receipt_confirmed_at = case
          when p_confirm_receipt and v_receipt_requested and v_receipt_confirmed_at is null
            then now()
          else receipt_confirmed_at
        end
    where id = p_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('request', p_request_id, p_request_id, 'response_saved',
      jsonb_build_object('done_date', p_done_date, 'done_time', p_done_time, 'reminder_enabled', p_reminder_enabled, 'overdue_reminder_enabled', p_overdue_reminder_enabled, 'reminder_day_of_enabled', p_reminder_day_of_enabled, 'confirm_receipt', p_confirm_receipt, 'via', 'recipient_session'));
end;
$$;

revoke all on function public.set_response_done_as_recipient(uuid, date, time, boolean, boolean, boolean, boolean) from public;
grant execute on function public.set_response_done_as_recipient(uuid, date, time, boolean, boolean, boolean, boolean) to authenticated;
