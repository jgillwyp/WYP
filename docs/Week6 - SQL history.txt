-- ============================================================================
-- WYP migration 029 — get_received_print_detail(): full Dialog + Attachments
-- for the Received print report
-- DRAFTED 2026-08-15 — NOT YET CONFIRMED RUN.
--
-- Context: the owner's own xlsx print mockups (Main Screen sections - Sent/
-- Received/ToDos) show each record's full Dialog thread (every Question/
-- Answer/Comment, in full) and full Attachments/Locations list under its
-- description, not just an icon indicating either exists. Sent and ToDos
-- already load dialog/attachments as PostgREST count embeds only
-- (dialog(count), attachments(count) — MainScreen.tsx's own .select()
-- strings), same as Received's own dialog_count/attachment_count (migration
-- 027) — none of the three sections has ever loaded full Dialog/Attachment
-- *content* for the Main Screen list, only counts, since the on-screen rows
-- only ever needed an icon, not the text.
--
-- Deliberately NOT switching the main list queries themselves to load full
-- content by default — that would add real Dialog-thread and Attachment-
-- metadata payload weight to every ordinary Main Screen load, for content
-- the on-screen UI never displays. Instead, the full content is fetched only
-- at the moment Print is actually clicked, for just the ids currently on
-- screen (see MainScreen.tsx's loadPrintDetail/startPrint).
--
-- Sent and ToDos can do this fetch as a plain client-side query — both are
-- rows the signed-in owner already has SELECT access to on public.dialog/
-- public.attachments under existing RLS (migration 025: "SELECT is
-- owner-only"). Received cannot: a Received row's dialog/attachments belong
-- to a different owner, invisible to the recipient's own RLS the same way
-- Category is (CLAUDE.md's Entitlements section) — this function is that
-- gap's fix, parallel to get_received_requests() (migration 012) itself,
-- keyed by the same contacts.email match against the caller's own session
-- email (auth.jwt() ->> 'email'), not a new access model.
--
-- Returns one row per matched request id (silently drops any id in p_ids
-- that isn't actually a Received request for the caller — same "return
-- nothing rather than error" posture as every other token/session-scoped
-- function here), with dialog and attachments each as a jsonb array so one
-- round trip covers every row on the current Received print. Field lists
-- match RequestDetailForm.tsx's own dialog select (id, kind, body, who,
-- replies_to_id) and the attachments API route's own select (id, kind,
-- file_name, reference_url, reference_note) — no new fields invented.
-- ============================================================================

begin;

create or replace function public.get_received_print_detail(p_ids uuid[])
returns table (
  request_id uuid,
  dialog jsonb,
  attachments jsonb
)
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
begin
  if v_email = '' or p_ids is null or array_length(p_ids, 1) is null then
    return; -- no email claim, or nothing asked for
  end if;

  return query
    select
      r.id as request_id,
      coalesce((
        select jsonb_agg(
                 jsonb_build_object(
                   'id', d.id, 'kind', d.kind, 'body', d.body,
                   'who', d.who, 'replies_to_id', d.replies_to_id
                 ) order by d.id
               )
        from public.dialog d
        where d.request_id = r.id
      ), '[]'::jsonb) as dialog,
      coalesce((
        select jsonb_agg(
                 jsonb_build_object(
                   'id', a.id, 'kind', a.kind, 'file_name', a.file_name,
                   'reference_url', a.reference_url, 'reference_note', a.reference_note
                 ) order by a.created_at
               )
        from public.attachments a
        where a.request_id = r.id and a.deleted_at is null
      ), '[]'::jsonb) as attachments
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = any(p_ids)
      and lower(trim(c.email)) = v_email;
end;
$$;

revoke all on function public.get_received_print_detail(uuid[]) from public;
grant execute on function public.get_received_print_detail(uuid[]) to authenticated;

commit;

-- WYP migration 030 — get_contact_request_counts(): per-contact Sent/Rec'd
-- counts for the new Contacts print report.
-- DRAFTED 2026-08-15 — CONFIRMED RUN by the owner 2026-08-15.
--
-- Owner, asked to pick between two readings of "Sent"/"Rec'd" on his own
-- Contacts print mockup: "Sent = requests I sent them; Rec'd = requests
-- they sent me" — the same vocabulary the app's own Main Screen Sent/
-- Received sections already use, not "how many of my Sent requests to them
-- are Done."
--
-- Sent is trivial and needs no new privilege — a plain count of my own
-- requests grouped by contact_id, already fully covered by existing RLS
-- (requests: owner-only). Rec'd is the hard half: get_received_requests()
-- (migration 012) already knows how to match "a Request sent to me" via
-- contacts.email on the *sender's* own contacts table equaling my session
-- email, but it never needed to know the sender's own real account email,
-- only their display name (owner_name). To attribute a Received Request
-- back to *which one of my own contacts* sent it, this function looks up
-- the sender's auth.users.email (their actual login email, not a display
-- name) and matches it against my own contacts.email. Reads auth.users
-- directly inside a SECURITY DEFINER function — same precedent and
-- reasoning as can_create_account() (migration 015): only this function's
-- owner role can see auth.users; a caller only ever gets integer counts
-- back, never anyone's raw email address.
--
-- Returns one row per one of the caller's own contacts (zero-filled counts
-- where there's no match), scoped entirely to the caller via auth.uid()/
-- auth.jwt() — nothing here is reachable by anon.
-- ============================================================================

begin;

create or replace function public.get_contact_request_counts()
returns table (
  contact_id uuid,
  sent_count bigint,
  received_count bigint
)
language plpgsql
security definer
set search_path = public, auth, extensions
as $$
declare
  v_uid uuid := auth.uid();
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
begin
  if v_uid is null then
    return; -- no session; nothing to compute
  end if;

  return query
    select
      c.id as contact_id,
      coalesce(sent.cnt, 0) as sent_count,
      coalesce(recv.cnt, 0) as received_count
    from public.contacts c
    left join (
      select r.contact_id, count(*) as cnt
      from public.requests r
      where r.owner_id = v_uid
      group by r.contact_id
    ) sent on sent.contact_id = c.id
    left join (
      -- Every Request actually received by me, then rolled up by the real
      -- sender email — mirrors get_received_requests()'s own match, plus
      -- one extra join to auth.users to get an email worth matching against
      -- my own contacts.email (owner_name/profiles alone isn't unique or
      -- guaranteed to match how I saved this person).
      select lower(trim(au.email)) as sender_email, count(*) as cnt
      from public.requests r2
      join public.contacts c2 on c2.id = r2.contact_id
      join auth.users au on au.id = r2.owner_id
      where v_email <> '' and lower(trim(c2.email)) = v_email
      group by lower(trim(au.email))
    ) recv on recv.sender_email = lower(trim(c.email))
    where c.owner_id = v_uid;
end;
$$;

revoke all on function public.get_contact_request_counts() from public;
grant execute on function public.get_contact_request_counts() to authenticated;

commit;

-- ----------------------------------------------------------------------------
-- get_received_request — add received_archived_at, deferred from migration
-- 028 ("revisit if a future Un-Archive-from-Detail affordance needs it").
-- returns jsonb, so a plain CREATE OR REPLACE is safe (no OUT-parameter row
-- shape to fight, unlike get_received_requests()'s own RETURNS TABLE).
-- Byte-identical to the migration 020 version otherwise.
-- ----------------------------------------------------------------------------
create or replace function public.get_received_request(p_request_id uuid)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.received_archived_at,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         c.email as contact_email
    into v_request
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    left join public.profiles p on p.id = r.owner_id
    where r.id = p_request_id;

  if v_request.id is null or v_email = '' or lower(trim(v_request.contact_email)) <> v_email then
    raise exception 'Request not found.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('request', v_request.id, v_request.id, 'viewed_by_recipient', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'received_archived_at', v_request.received_archived_at,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_received_request(uuid) from public;
grant execute on function public.get_received_request(uuid) to authenticated;

-- ----------------------------------------------------------------------------
-- set_response_done_as_recipient — clear received_archived_at in the same
-- UPDATE whenever p_done_date is null (un-archive-on-clear); preserved
-- unchanged otherwise. Byte-identical to the migration 012 version apart
-- from that one added case expression.
-- ----------------------------------------------------------------------------
create or replace function public.set_response_done_as_recipient(
  p_request_id uuid,
  p_done_date date,
  p_done_time time
)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request_id uuid;
begin
  select r.id into v_request_id
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = p_request_id
      and v_email <> ''
      and lower(trim(c.email)) = v_email;

  if v_request_id is null then
    raise exception 'Request not found.';
  end if;

  update public.requests
    set done_date = p_done_date,
        done_time = p_done_time,
        received_archived_at = case when p_done_date is null then null else received_archived_at end
    where id = v_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values (
      'request', v_request_id, v_request_id, 'response_done_updated',
      jsonb_build_object('done_date', p_done_date, 'done_time', p_done_time, 'via', 'recipient_session')
    );
end;
$$;

revoke all on function public.set_response_done_as_recipient(uuid, date, time) from public;
grant execute on function public.set_response_done_as_recipient(uuid, date, time) to authenticated;

-- ============================================================================
-- WYP migration 033 — cron_issue_request_link(): service_role-only response
-- link minting for the cron notification system.
-- DRAFTED 2026-08-17 — NOT YET CONFIRMED RUN.
--
-- app/api/cron/tick/route.ts (built alongside this migration) needs to embed
-- a working /r/[token] response link in every Reminder and Overdue email it
-- sends — the same link Request Detail's own "Get Response Link" band used
-- to expose manually (removed 2026-08-14) and CreateRequestForm.tsx already
-- mints automatically on Send. That existing function, issue_request_link
-- (migration 008), is owner-only by design: it checks auth.uid() = the
-- Request's own owner_id. A cron run has no session — auth.uid() is null —
-- so it cannot call issue_request_link at all, and must not: relaxing that
-- function's own check would let any authenticated caller mint a link for
-- any Request, a real privilege escalation.
--
-- cron_issue_request_link is a separate function, not a relaxed version of
-- issue_request_link — same token/hash/expiry mechanics (30-day expiry,
-- salted SHA-256 hash, existing events logging), but with NO auth.uid()
-- check at all. Safe only because it is never granted to authenticated or
-- anon — only to service_role, which app/api/cron/tick/route.ts alone holds
-- (SUPABASE_SERVICE_ROLE_KEY, server-only env var, same posture as
-- app/api/attachments/*/route.ts). A caller who already has service_role
-- can bypass RLS and read/write the requests table directly regardless, so
-- this function grants no new capability to anyone who didn't already have
-- it — it exists only so the cron route doesn't have to reimplement
-- token-hash generation in JS and risk a byte-level mismatch against
-- get_request_by_token's own `digest(p_token, 'sha256')` comparison.
--
-- Always mints a fresh token and overwrites link_token_hash, exactly like
-- issue_request_link itself already does unconditionally on every call
-- (migration 008 has no "reuse if still valid" branch either) — deliberate,
-- not an oversight. Only the hash is ever persisted (CLAUDE.md's own
-- "store digest(token,'sha256') not the token" rule), so a still-valid
-- earlier link's raw value can never be recovered to re-embed in a second
-- email; the only options are mint-fresh-and-supersede (what this does, and
-- what the app has always done) or leave a stale link in an older email
-- forever un-refreshed. A recipient who already clicked an earlier email's
-- link and is mid-session on /r/[token] is unaffected either way — that
-- page holds the token client-side for the rest of its own visit,
-- independent of what the database's own current hash is.
-- ============================================================================

begin;

create or replace function public.cron_issue_request_link(p_request_id uuid)
returns text
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_owner uuid;
  v_token text;
begin
  select owner_id into v_owner from public.requests where id = p_request_id;

  if v_owner is null then
    raise exception 'Not found.';
  end if;

  v_token := encode(gen_random_bytes(32), 'hex');

  update public.requests
    set link_token_hash = digest(v_token, 'sha256'),
        link_expires_at = now() + interval '30 days',
        link_revoked_at = null
    where id = p_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', p_request_id, p_request_id, 'issued', '{"via": "cron"}'::jsonb);

  return v_token;
end;
$$;

create or replace function public.can_toggle_tier()
returns boolean
language plpgsql
security definer
set search_path = public, auth
as $$
declare
  v_gate_enabled boolean;
  v_email text;
  v_allowlisted boolean;
begin
  select coalesce(
    (select value from app_settings where key = 'tier_toggle_gate_enabled'),
    true
  ) into v_gate_enabled;

  if not v_gate_enabled then
    return true;
  end if;

  select lower(trim(email)) into v_email from auth.users where id = auth.uid();

  if v_email is null then
    return false;
  end if;

  select exists(
    select 1 from tier_toggle_allowlist where lower(email) = v_email
  ) into v_allowlisted;

  return v_allowlisted;
end;
$$;

revoke all on function public.can_toggle_tier() from public;
grant execute on function public.can_toggle_tier() to authenticated;

create or replace function public.set_tier_for_testing(p_tier text)
returns void

security definer
set search_path = public, auth
as $$

  update public.profiles
    set tier = p_tier

end;
$$;

create or replace function public.get_request_by_token(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.reminder_enabled,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         ct.display_name as contact_name,
         r.link_expires_at, r.link_revoked_at
    into v_request
    from public.requests r
    left join public.profiles p on p.id = r.owner_id
    left join public.contacts ct on ct.id = r.contact_id
    where r.link_token_hash = v_hash;

  if v_request.id is null
     or v_request.link_revoked_at is not null
     or v_request.link_expires_at < now()
  then
    raise exception 'This link is no longer valid.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', v_request.id, v_request.id, 'viewed', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'reminder_enabled', v_request.reminder_enabled,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'contact_name', v_request.contact_name,
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_request_by_token(text) from public;
grant execute on function public.get_request_by_token(text) to anon, authenticated;

create or replace function public.set_response_done_as_recipient(
  p_request_id uuid,
  p_done_date date,
  p_done_time time,
  p_reminder_enabled boolean default null
)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request_id uuid;

  select r.id into v_request_id
    from public.requests r
    join public.contacts c on c.id = r.contact_id

      'request', v_request_id, v_request_id, 'response_done_updated',
      jsonb_build_object('done_date', p_done_date, 'done_time', p_done_time, 'reminder_enabled', p_reminder_enabled, 'via', 'recipient_session')
    );
end;

-- ============================================================================
-- WYP migration 038 — Repeat feature (recurring Requests and ToDos)
-- DRAFTED 2026-08-21 — CONFIRMED RUN by the owner 2026-08-21.
--
-- Jim's own recurrence-method design ("WYP Repeat design.docx", uploaded
-- 2026-08-21) — a Request or ToDo can repeat on a Day/Week/Month/Year
-- schedule, anchored to its own Due Date (never Done Date — Jim's explicit
-- instruction: "the Due Date should be the determinant as to the generation
-- of the next Request"). See app/src/lib/repeatRule.ts for the pure
-- recurrence math and descriptive-text builder shared by every screen.
--
-- requests.repeat_rule (jsonb) — the whole RepeatRule object (type,
-- interval, weekdaysOnly, monthMode, stopType, stopDate, stopCount — see
-- repeatRule.ts's own RepeatRule type for the exact shape). Null means "not
-- repeating." Copied verbatim onto every generated successor so the chain
-- continues without re-reading the original.
--
-- requests.repeat_occurrence_index (integer) — 1-based position in the
-- series; the very first Request/ToDo (the one the user configured Repeat
-- on) is 1, its first generated successor is 2, and so on. Only meaningful
-- when repeat_rule is set; needed for the "stops after N occurrences"
-- condition, which can't be answered by counting rows — nothing links a
-- series together beyond each row's own repeat_rule copy (no series id,
-- since nothing in this feature needs to list a whole series, only ever
-- the current and next occurrence).
--
-- requests.repeat_next_generated_at (timestamptz) — idempotency marker, same
-- convention as reminder_sent_at/overdue_notified_at (migration 032): set by
-- the cron job (app/api/cron/tick/route.ts, new Phase E) the moment it has
-- processed this row's own Due-Date-arrival for repeat-generation purposes,
-- whether that produced a successor or hit a stop condition. Prevents a
-- second cron run in the same hour from double-inserting.
--
-- attachments.carry_into_repeats (boolean) — Jim's own design for carrying
-- Attachments/Locations into a repeating series: at Send/Save time (or
-- later, via Edit Repeat), the sender is shown the existing Attachments/
-- Locations list with checkboxes and told Dialog is never carried forward,
-- only whichever items they check. A boolean on the row itself, not a
-- separate join table or an array column on requests — the cron job's own
-- generation step just selects `where carry_into_repeats = true` on the
-- occurrence it's about to supersede, and stamps the same flag on every
-- duplicate it creates, so the selection propagates down the whole chain
-- with no re-asking.
--
-- Real 'file' Attachments are duplicated as a fresh Storage object per
-- generated occurrence (app/api/cron/tick/route.ts uses storage.copy()),
-- never shared across multiple `attachments` rows — matches this table's
-- existing hard-delete-per-row model (migration 025's own header comment)
-- and avoids the complexity of reference-counted deletes. Accepted
-- trade-off, confirmed with Jim: an unbounded "every day, stops never"
-- repeat with a carried file keeps duplicating that file's bytes into every
-- occurrence indefinitely — a real but currently acceptable storage cost at
-- this app's testing scale.
--
-- Archived Requests/ToDos are exempt from repeat generation, same posture
-- CLAUDE.md/the cron system already takes for every other notification
-- (migration 032's own "archived items are exempt from every notification"
-- rule) — an archived row's Repeat control is already greyed/non-modifiable
-- in the UI (Jim's own spec), so the cron job honors that by simply never
-- generating a successor for an archived row, rather than generating one
-- silently behind the scenes.
-- ============================================================================

begin;

alter table public.requests
  add column if not exists repeat_rule jsonb,
  add column if not exists repeat_occurrence_index integer,
  add column if not exists repeat_next_generated_at timestamptz;

comment on column public.requests.repeat_rule is
  'Recurrence rule (Day/Week/Month/Year + interval + stop condition), or null if this Request/ToDo does not repeat. Shape defined by RepeatRule in app/src/lib/repeatRule.ts. Copied verbatim onto every generated successor.';
comment on column public.requests.repeat_occurrence_index is
  '1-based position in this row''s own repeat series (1 = the original, 2 = its first generated successor, ...). Only meaningful when repeat_rule is set.';
comment on column public.requests.repeat_next_generated_at is
  'Idempotency marker for the cron job''s Repeat-generation phase — set once this row''s own Due Date has been processed (successor generated, or a stop condition was hit), whether or not a new row was actually inserted.';

alter table public.attachments
  add column if not exists carry_into_repeats boolean not null default false;

comment on column public.attachments.carry_into_repeats is
  'Set at Send/Save time (or later via Edit Repeat) when the sender selects this Attachment/Location to carry into every future occurrence of a repeating Request/ToDo. The cron job duplicates every row with this flag set onto each newly generated occurrence, propagating the flag onto the duplicate too, so the selection continues down the whole chain without re-asking.';

-- repeat_rule/repeat_occurrence_index/repeat_next_generated_at need no new
-- RLS/grants — plain requests columns, already covered by that table's
-- existing owner-scoped policies (migration 002/003).
--
-- carry_into_repeats is different: migration 025 deliberately created NO
-- update policy on attachments at all ("an attachment is added or removed,
-- never edited in place" — that migration's own header comment), so a
-- direct client toggle of this new column would silently fail RLS with no
-- policy to admit it. Rather than open a general UPDATE (which would
-- contradict that same append-only design), this adds a narrow one scoped
-- to a single column — same two-part shape (RLS policy + column-level
-- GRANT) migration 013 already established for profiles.time_zone: the RLS
-- policy admits an UPDATE on any row belonging to the caller's own
-- Request/ToDo, and the GRANT then restricts that permission to touching
-- only carry_into_repeats — an owner still cannot rename, resize, or
-- otherwise edit an existing attachment in place.
create policy "attachments: owners update carry_into_repeats"
  on public.attachments for update to authenticated
  using (request_id in (select id from public.requests where owner_id = auth.uid()))
  with check (request_id in (select id from public.requests where owner_id = auth.uid()));

grant update (carry_into_repeats) on public.attachments to authenticated;

commit;

-- ----------------------------------------------------------------------------
-- get_request_by_token — add reminder_day_of_enabled/reminder_day_of_sent_at
-- to the jsonb payload.
-- ----------------------------------------------------------------------------
create or replace function public.get_request_by_token(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.reminder_enabled, r.reminder_sent_at, r.overdue_reminder_enabled, r.reminder_day_of_enabled,
         r.reminder_day_of_sent_at, r.repeat_rule,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         ct.display_name as contact_name,
         r.link_expires_at, r.link_revoked_at
    into v_request
    from public.requests r
    left join public.profiles p on p.id = r.owner_id
    left join public.contacts ct on ct.id = r.contact_id
    where r.link_token_hash = v_hash;

  if v_request.id is null
     or v_request.link_revoked_at is not null
     or v_request.link_expires_at < now()
  then
    raise exception 'This link is no longer valid.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', v_request.id, v_request.id, 'viewed', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'reminder_enabled', v_request.reminder_enabled,
    'reminder_sent_at', v_request.reminder_sent_at,
    'overdue_reminder_enabled', v_request.overdue_reminder_enabled,
    'reminder_day_of_enabled', v_request.reminder_day_of_enabled,
    'reminder_day_of_sent_at', v_request.reminder_day_of_sent_at,
    'repeat_rule', v_request.repeat_rule,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'contact_name', v_request.contact_name,
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_request_by_token(text) from public;
grant execute on function public.get_request_by_token(text) to anon, authenticated;

-- ----------------------------------------------------------------------------
-- get_received_request — same addition, signed-in recipient path.
-- ----------------------------------------------------------------------------
create or replace function public.get_received_request(p_request_id uuid)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.received_archived_at, r.reminder_enabled, r.reminder_sent_at, r.overdue_reminder_enabled,
         r.reminder_day_of_enabled, r.reminder_day_of_sent_at, r.repeat_rule,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         c.email as contact_email
    into v_request
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    left join public.profiles p on p.id = r.owner_id
    where r.id = p_request_id;

  if v_request.id is null or v_email = '' or lower(trim(v_request.contact_email)) <> v_email then
    raise exception 'Request not found.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('request', v_request.id, v_request.id, 'viewed_by_recipient', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'received_archived_at', v_request.received_archived_at,
    'reminder_enabled', v_request.reminder_enabled,
    'reminder_sent_at', v_request.reminder_sent_at,
    'overdue_reminder_enabled', v_request.overdue_reminder_enabled,
    'reminder_day_of_enabled', v_request.reminder_day_of_enabled,
    'reminder_day_of_sent_at', v_request.reminder_day_of_sent_at,
    'repeat_rule', v_request.repeat_rule,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_received_request(uuid) from public;
grant execute on function public.get_received_request(uuid) to authenticated;

-- ----------------------------------------------------------------------------
-- set_response_done_by_token — gains p_reminder_day_of_enabled, same
-- coalesce-against-existing-column pattern as p_reminder_enabled/
-- p_overdue_reminder_enabled.
-- ----------------------------------------------------------------------------
create or replace function public.set_response_done_by_token(
  p_token text,
  p_done_date date,
  p_done_time time,
  p_reminder_enabled boolean default null,
  p_overdue_reminder_enabled boolean default null,
  p_reminder_day_of_enabled boolean default null
)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request_id uuid;
begin
  select r.id into v_request_id
    from public.requests r
    where r.link_token_hash = v_hash
      and r.link_revoked_at is null
      and r.link_expires_at > now();

  if v_request_id is null then
    raise exception 'This link is no longer valid.';
  end if;

  update public.requests
    set done_date = p_done_date,
        done_time = p_done_time,
        reminder_enabled = coalesce(p_reminder_enabled, reminder_enabled),
        overdue_reminder_enabled = coalesce(p_overdue_reminder_enabled, overdue_reminder_enabled),
        reminder_day_of_enabled = coalesce(p_reminder_day_of_enabled, reminder_day_of_enabled)
    where id = v_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('request', v_request_id, v_request_id, 'response_saved',
      jsonb_build_object('done_date', p_done_date, 'done_time', p_done_time, 'reminder_enabled', p_reminder_enabled, 'overdue_reminder_enabled', p_overdue_reminder_enabled, 'reminder_day_of_enabled', p_reminder_day_of_enabled));
end;
$$;

revoke all on function public.set_response_done_by_token(text, date, time, boolean, boolean, boolean) from public;
grant execute on function public.set_response_done_by_token(text, date, time, boolean, boolean, boolean) to anon, authenticated;

-- ----------------------------------------------------------------------------
-- set_response_done_as_recipient — same addition, signed-in recipient path.
-- ----------------------------------------------------------------------------
create or replace function public.set_response_done_as_recipient(
  p_request_id uuid,
  p_done_date date,
  p_done_time time,
  p_reminder_enabled boolean default null,
  p_overdue_reminder_enabled boolean default null,
  p_reminder_day_of_enabled boolean default null
)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_contact_email text;
begin
  select lower(trim(c.email)) into v_contact_email
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = p_request_id;

  if v_email = '' or v_contact_email is null or v_contact_email <> v_email then
    raise exception 'Request not found.';
  end if;

  update public.requests
    set done_date = p_done_date,
        done_time = p_done_time,
        received_archived_at = case when p_done_date is null then received_archived_at else null end,
        reminder_enabled = coalesce(p_reminder_enabled, reminder_enabled),
        overdue_reminder_enabled = coalesce(p_overdue_reminder_enabled, overdue_reminder_enabled),
        reminder_day_of_enabled = coalesce(p_reminder_day_of_enabled, reminder_day_of_enabled)
    where id = p_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('request', p_request_id, p_request_id, 'response_saved',
      jsonb_build_object('done_date', p_done_date, 'done_time', p_done_time, 'reminder_enabled', p_reminder_enabled, 'overdue_reminder_enabled', p_overdue_reminder_enabled, 'reminder_day_of_enabled', p_reminder_day_of_enabled, 'via', 'recipient_session'));
end;
$$;

revoke all on function public.set_response_done_as_recipient(uuid, date, time, boolean, boolean, boolean) from public;
grant execute on function public.set_response_done_as_recipient(uuid, date, time, boolean, boolean, boolean) to authenticated;

-- ----------------------------------------------------------------------------
-- get_request_by_token — add owner_request_reminders_enabled to the jsonb
-- payload, same coalesce-true-if-null pattern already used for
-- owner_request_time_enabled (a null only happens if the owner's own
-- profiles row is somehow missing, which should never occur in practice).
-- jsonb-returning, so a plain create-or-replace is safe -- no OUT-parameter
-- shape to worry about (see migration 017's own header for why that
-- constraint doesn't apply to jsonb functions).
-- ----------------------------------------------------------------------------
create or replace function public.get_request_by_token(p_token text)
returns jsonb
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_hash bytea := digest(p_token, 'sha256');
  v_request record;
  v_result jsonb;
begin
  select r.id, r.description, r.created_at, r.due_date, r.due_time, r.done_date, r.done_time,
         r.reminder_enabled, r.reminder_sent_at, r.overdue_reminder_enabled, r.reminder_day_of_enabled,
         r.reminder_day_of_sent_at, r.repeat_rule,
         p.display_name as owner_name, p.tier as owner_tier,
         p.request_time_enabled as owner_request_time_enabled,
         p.request_reminders_enabled as owner_request_reminders_enabled,
         ct.display_name as contact_name,
         r.link_expires_at, r.link_revoked_at
    into v_request
    from public.requests r
    left join public.profiles p on p.id = r.owner_id
    left join public.contacts ct on ct.id = r.contact_id
    where r.link_token_hash = v_hash;

  if v_request.id is null
     or v_request.link_revoked_at is not null
     or v_request.link_expires_at < now()
  then
    raise exception 'This link is no longer valid.';
  end if;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values ('link', v_request.id, v_request.id, 'viewed', '{}'::jsonb);

  select jsonb_build_object(
    'id', v_request.id,
    'description', v_request.description,
    'created_at', v_request.created_at,
    'due_date', v_request.due_date,
    'due_time', v_request.due_time,
    'done_date', v_request.done_date,
    'done_time', v_request.done_time,
    'reminder_enabled', v_request.reminder_enabled,
    'reminder_sent_at', v_request.reminder_sent_at,
    'overdue_reminder_enabled', v_request.overdue_reminder_enabled,
    'reminder_day_of_enabled', v_request.reminder_day_of_enabled,
    'reminder_day_of_sent_at', v_request.reminder_day_of_sent_at,
    'repeat_rule', v_request.repeat_rule,
    'owner_name', v_request.owner_name,
    'owner_tier', v_request.owner_tier,
    'owner_request_time_enabled', coalesce(v_request.owner_request_time_enabled, true),
    'owner_request_reminders_enabled', coalesce(v_request.owner_request_reminders_enabled, true),
    'contact_name', v_request.contact_name,
    'dialog', coalesce(
      jsonb_agg(
        jsonb_build_object(
          'id', d.id, 'kind', d.kind, 'body', d.body, 'who', d.who,
          'created_at', d.created_at, 'replies_to_id', d.replies_to_id
        ) order by d.created_at desc
      ) filter (where d.id is not null),
      '[]'::jsonb
    )
  )
  into v_result
  from public.dialog d
  where d.request_id = v_request.id;

  return v_result;
end;
$$;

revoke all on function public.get_request_by_token(text) from public;
grant execute on function public.get_request_by_token(text) to anon, authenticated;

-- Received has no such path — RLS on requests is owner-only, and the
-- recipient isn't the owner, which is exactly why archive_received_request()
-- (migration 028) had to be a SECURITY DEFINER function in the first place.
-- unarchive_received_request() is its direct mirror: identical email-match-
-- through-contacts guard, identical single-row-per-call shape (called once
-- per id from ArchiveForm.tsx, same as its Archive-side counterpart), only
-- difference is `set received_archived_at = null` instead of `= now()`, and
-- an `'unarchived_by_recipient'` events action instead of `'archived_by_
-- recipient'`. Body copied from migration 028's own archive_received_

-- rather than reconstructed from memory (this project's own established
-- practice after an earlier session caught a guessed function body drifting

begin;

-- ----------------------------------------------------------------------------
-- unarchive_received_request — the signed-in recipient's UnArchive action on
-- their own Received copy of a Request. Mirrors archive_received_request()
-- (migration 028) exactly; only sets received_archived_at, touches nothing
-- else on the row.
-- ----------------------------------------------------------------------------
create or replace function public.unarchive_received_request(p_request_id uuid)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_email text := lower(trim(coalesce(auth.jwt() ->> 'email', '')));
  v_request_id uuid;
begin
  select r.id into v_request_id
    from public.requests r
    join public.contacts c on c.id = r.contact_id
    where r.id = p_request_id
      and v_email <> ''
      and lower(trim(c.email)) = v_email;

  if v_request_id is null then
    raise exception 'Request not found.';
  end if;

  update public.requests
    set received_archived_at = null
    where id = v_request_id;

  insert into public.events (subject_type, subject_id, request_id, action, detail)
    values (
      'request', v_request_id, v_request_id, 'unarchived_by_recipient',
      jsonb_build_object('via', 'recipient_session')
    );
end;
$$;

grant execute on function public.unarchive_received_request(uuid) to authenticated;

set search_path = public, auth

  if not public.can_toggle_tier() then

-- WYP migration 048 — fold existing ToDo Locations into their own
-- Description; retire kind='reference' attachments rows
-- DRAFTED 2026-08-26 — NOT YET CONFIRMED RUN.
--
-- Context: Jim's own Request/ToDo-symmetry proposal (2026-08-26 decisions
-- log) replaces ToDo Locations outright with real file Attachments (the
-- exact mechanism Requests already use — see AttachmentsPanel.tsx's own
-- mode='file'/'reference' split, which needs no schema change to serve a
-- ToDo, since a ToDo is just a public.requests row with contact_id = null,
-- and every attachments RLS policy/API route is already keyed on
-- ownership alone, never on Request-vs-ToDo). Before ToDo Detail's
-- AttachmentsPanel switches from mode='reference' to mode='file', any
-- Locations a ToDo already has need to survive the switch in some form —
-- Jim's own instruction: "very little usage... collect any Locations and
-- append them as text to the Description as ' -- Location(s): xxxxx,
-- yyyyyy, zzzzz'." Requests never had Locations (mode='reference' has only
-- ever applied to ToDos, per AttachmentsPanel.tsx's own header comment),
-- so this only ever touches ToDo rows in practice, even though the query
-- itself is written generically off attachments.kind rather than filtering
-- requests.contact_id — kind='reference' already implies "was a Location."
--
-- Per-request_id entries are joined with ", ", each entry rendered as
-- "note: value" when a Location has both a Description and a Location
-- value, or just whichever of the two is present otherwise — Jim confirmed
-- this format needs no change from the first draft. Requests with no
-- Locations at all are untouched (the loop simply never visits them).
--
-- Runs before the application code deploys that stops writing/reading
-- kind='reference' rows at all — once TodoDetailForm.tsx and
-- CreateTodoForm.tsx switch to mode='file', nothing in the app inserts a
-- 'reference' row again, so deleting every existing one here is a clean,
-- one-time cutover, not an ongoing behavior change.
-- ============================================================================

begin;

do $$
declare
  r record;
  locs text;
begin
  for r in
    select request_id,
           string_agg(
             coalesce(nullif(reference_note, '') || ': ', '') ||
             coalesce(reference_url, reference_note, ''),
             ', ' order by created_at
           ) as locs
    from public.attachments
    where kind = 'reference'
    group by request_id
  loop
    if r.locs is not null and length(trim(r.locs)) > 0 then
      update public.requests
        set description = coalesce(description, '') || ' -- Location(s): ' || r.locs
        where id = r.request_id;
    end if;
  end loop;
end $$;

delete from public.attachments where kind = 'reference';

commit;
