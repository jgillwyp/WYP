-- July 30 started

   create table contacts (
     id          uuid primary key default gen_random_uuid(),
     owner_id    uuid not null references auth.users(id) on delete cascade,
     first_name  text,
     last_name   text,
     email       text not null,
     phone       text,
     send_by     text not null default 'email',   -- 'email' | 'text'
     notes       text,
     created_at  timestamptz not null default now()
   );
   
select last_name from contacts
INSERT INTO contacts (owner_id,first_name,last_name,email,phone,send_by)
VALUES ('480ab1de-2c16-479a-ae1f-ae0daa4c21ba','Mary','Jones','mjones@somewhere.com','999-333-5812','email');

CREATE POLICY "Allow individual insert" 
ON contacts 
FOR INSERT 
TO authenticated 
WITH CHECK (auth.uid() = owner_id);
SELECT id FROM auth.users LIMIT 1;
select * from contacts

alter table contacts alter column owner_id set default auth.uid();

select auth.uid()

-- to use within app
create or replace function public.test_auth_context()
returns json
language sql
security definer
as $$
  select json_build_object(
    'uid', auth.uid(),
    'jwt', auth.jwt()
  );
$$;
