-- ============================================================================
-- WYP migration 003 — requests (unified Request/ToDo), categories
-- Run 2026-08-06 in the Supabase SQL editor.
--
-- RUN NOTE: the categories table below originally used a table-level
-- `unique (owner_id, lower(name))` constraint, which Postgres rejects —
-- `ERROR 42601: syntax error at or near "("`. Table-level UNIQUE only accepts
-- plain columns, not expressions. Fixed by dropping the inline constraint and
-- adding a separate unique index (categories_owner_name_ci_idx) after the
-- table, which is the correct form for a case-insensitive uniqueness rule.
-- The block below is the corrected, as-run version.
-- ============================================================================

begin;

-- ----------------------------------------------------------------------------
-- 1. categories — owner-scoped labels, 20-cap enforced at the app layer
--    (matches contacts' existing owner-only shape; nothing new to explain here)
-- ----------------------------------------------------------------------------
create table if not exists public.categories (
  id          uuid primary key default gen_random_uuid(),
  owner_id    uuid not null references auth.users(id) on delete cascade,
  name        text not null,
  created_at  timestamptz not null default now()
);

-- table-level `unique (...)` only accepts plain columns, not expressions like
-- lower(name) — a unique index is the correct form here.
create unique index if not exists categories_owner_name_ci_idx
  on public.categories (owner_id, lower(name));

alter table public.categories enable row level security;

create policy "categories: owners select own"
  on public.categories for select to authenticated
  using (owner_id = auth.uid());
create policy "categories: owners insert own"
  on public.categories for insert to authenticated
  with check (owner_id = auth.uid());
create policy "categories: owners update own"
  on public.categories for update to authenticated
  using (owner_id = auth.uid()) with check (owner_id = auth.uid());
create policy "categories: owners delete own"
  on public.categories for delete to authenticated
  using (owner_id = auth.uid());

-- ----------------------------------------------------------------------------
-- 2. requests — PRD §2.5: "ToDos are stored as Requests." One table.
--    contact_id null            -> ToDo (no recipient)
--    contact_id not null        -> Request
--    priority is meaningful only when contact_id is null (ToDo); PRD gives
--    Requests no Priority field
--    due_date/due_time required by the app for a Request, optional for a ToDo
-- ----------------------------------------------------------------------------
create table if not exists public.requests (
  id            uuid primary key default gen_random_uuid(),
  owner_id      uuid not null references auth.users(id) on delete cascade,
  contact_id    uuid references public.contacts(id) on delete set null,
  category_id   uuid references public.categories(id) on delete set null,

  description   text not null,                 -- 500-char app-wide convention (§2.4) — confirm
  priority      smallint check (priority in (1,2,3)),  -- 1 ASAP / 2 SOON / 3 LATER; ToDo only

  due_date      date,
  due_time      time,                           -- sender's local wall-clock (§2.7)
  done_date     date,
  done_time     time,

  created_at    timestamptz not null default now(),
  updated_at    timestamptz not null default now()
);

comment on column public.requests.contact_id is
  'Null = ToDo (PRD §2.5). Not null = Request, and this is the recipient.';
comment on column public.requests.priority is
  'ToDo-only per §2.1 core objects table. Requests have no Priority field.';

create index if not exists requests_owner_idx    on public.requests (owner_id);
create index if not exists requests_contact_idx  on public.requests (contact_id);
create index if not exists requests_category_idx on public.requests (category_id);

alter table public.requests enable row level security;

-- Owner (sender) only, for now — see the note above on recipient visibility.
create policy "requests: owners select own"
  on public.requests for select to authenticated
  using (owner_id = auth.uid());
create policy "requests: owners insert own"
  on public.requests for insert to authenticated
  with check (owner_id = auth.uid());
create policy "requests: owners update own"
  on public.requests for update to authenticated
  using (owner_id = auth.uid()) with check (owner_id = auth.uid());
create policy "requests: owners delete own"
  on public.requests for delete to authenticated
  using (owner_id = auth.uid());

create trigger requests_touch_updated_at
  before update on public.requests
  for each row execute function public.touch_updated_at();  -- reused from migration 002

commit;

-- ============================================================================
-- WYP migration 004 — dialog
-- Not yet run. Needed to back the Dialog field added to Create Request
-- (and, later, Create ToDo) — 2026-08-06.
--
-- request_id covers both Requests and ToDos, since ToDos are rows in
-- `requests` too (§2.5) — one dialog table, no per-object-type duplication.
--
-- `who` is a display-name SNAPSHOT at post time, not a live join to
-- `profiles`. Matches the immutable-fact reasoning already used for `events`
-- (migration 002): if the owner later renames themselves, a Dialog entry from
-- last month should still show the name it showed then, not silently rewrite
-- history. `author_user_id` is kept alongside for anything that does need a
-- live identity (recipient-side dialog, once that's built) but `who` is what
-- the UI reads for display.
--
-- No UPDATE or DELETE policy: append-only, same as `events` — editing or
-- retracting a past Question/Answer/Comment would falsify the thread.
--
-- `id` is a plain sequential bigint on purpose — it doubles as the sort key
-- alongside `created_at` when two entries land on the same day (decisions
-- log, 2026-08-05: "if internal IDs are sequentially assigned, they could be
-- used for this sort").
-- ============================================================================

begin;

create table if not exists public.dialog (
  id              bigint      generated always as identity primary key,
  request_id      uuid        not null references public.requests(id) on delete cascade,
  kind            text        not null default 'comment'
                              check (kind in ('question', 'answer', 'comment')),
  body            text        not null,
  who             text        not null,   -- display-name snapshot, see note above
  author_user_id  uuid        references auth.users(id) on delete set null,
  created_at      timestamptz not null default now()
);

comment on column public.dialog.who is
  'Display-name snapshot at post time, not a live join to profiles. A later name change must not rewrite what a past entry showed.';
comment on column public.dialog.id is
  'Sequential by design — the sort key, alongside created_at, when entries share a date. See decisions log 2026-08-05.';

create index if not exists dialog_request_idx on public.dialog (request_id, id);

alter table public.dialog enable row level security;

create policy "dialog: owners select own"
  on public.dialog for select to authenticated
  using (request_id in (select id from public.requests where owner_id = auth.uid()));

create policy "dialog: owners insert own"
  on public.dialog for insert to authenticated
  with check (
    request_id in (select id from public.requests where owner_id = auth.uid())
    and author_user_id = auth.uid()
  );

-- No update or delete policy — see the append-only note above.

commit;

-- WYP migration 005 — contacts.display_name
-- Not yet run. Needed before Add Contact / Create Request can be converted to
-- the single-Name-field UI decided 2026-08-07.
--
-- first_name/last_name are KEPT, not dropped — the owner's call: Create my
-- Free Account still collects both First and Last Name plus a Display Name
-- (unchanged), and a Request recipient who signs up gets their contact's
-- Display Name copied into that Display Name field, then must add at least a
-- First or Last Name of their own. Contacts and profiles are different
-- objects with different rules; contacts' first_name/last_name stay in the
-- table for possible later use but the app stops writing them going forward
-- — Add Contact becomes a single Name field that writes only display_name.
-- The few existing test rows keep whatever first_name/last_name they already
-- have; no further action needed on those values.
-- ============================================================================

begin;

alter table public.contacts add column if not exists display_name text;

-- Backfill: concatenate what's there today. A handful of rows have neither
-- name (email-only test inserts, see Week1 SQL history) — fall back to email
-- so the not-null/non-blank constraint below has something to hold onto.
update public.contacts
  set display_name = nullif(trim(concat_ws(' ', first_name, last_name)), '')
  where display_name is null;

update public.contacts
  set display_name = email
  where display_name is null or trim(display_name) = '';

alter table public.contacts
  alter column display_name set not null;

alter table public.contacts
  add constraint contacts_display_name_not_blank check (trim(display_name) <> '');

comment on column public.contacts.display_name is
  'The name the app shows and edits from 2026-08-07 forward — Add Contact writes only this column now.';
comment on column public.contacts.first_name is
  'Preserved but not populated by the app going forward (2026-08-07 decision) — kept for possible later use, not read by any current screen.';
comment on column public.contacts.last_name is
  'Preserved but not populated by the app going forward (2026-08-07 decision) — kept for possible later use, not read by any current screen.';

commit;

-- ============================================================================
-- WYP migration 006 — dialog.replies_to_id
-- Not yet run. Needed for the Add Dialog modal's "which Question" picker
-- (2026-08-07): an Answer can now point at the specific Question it answers,
-- rather than only being inferrable by its position in the list.
--
-- Nullable, and only ever set when kind = 'answer' — enforced by a CHECK,
-- not by splitting the table, since every other Dialog concern (RLS, sort,
-- who) is shared across all three kinds. Self-reference (an entry can't
-- reply to itself) is also a CHECK. What a CHECK cannot do — confirm the
-- referenced row is actually kind = 'question', or that it belongs to the
-- same request_id — is left to the app layer. A trigger could enforce both,
-- but nothing else in this schema uses a cross-row trigger (touch_updated_at
-- is row-local), and at personal-append-only-thread scale a wrong link is a
-- display bug, not a security or integrity hole worth the added moving part.
-- Flagging the trade-off rather than silently assuming it.
--
-- on delete set null (not cascade): if a Question is ever removed by some
-- future maintenance path, an Answer that pointed to it should keep existing
-- as a Dialog entry — just lose its "Re:" quote — not vanish with it. Dialog
-- has no delete policy today (append-only), so this is a forward-looking
-- default, not a currently-reachable path.
-- ============================================================================

begin;

alter table public.dialog
  add column if not exists replies_to_id bigint references public.dialog(id) on delete set null;

alter table public.dialog
  add constraint dialog_replies_to_only_on_answer
  check (kind = 'answer' or replies_to_id is null);

alter table public.dialog
  add constraint dialog_replies_to_not_self
  check (replies_to_id is null or replies_to_id <> id);

comment on column public.dialog.replies_to_id is
  'Which Question (dialog.id) this Answer responds to. Null for question/comment rows, and for an answer composed when only one Question was open (no picker shown — see the Add Dialog modal, 2026-08-07). Does not reorder the thread: display stays entry-order (id/created_at); this only drives the "Re: <question>" quote shown on an Answer row and which Questions still count as open.';

commit;

-- ============================================================================
-- WYP seed — demo Contacts, Sent Requests, and ToDos for the Main Screen
-- Not run yet. Written 2026-08-08 for the Main Screen React conversion —
-- "I would like to see the WYP app ... have a working main screen." Not a
-- schema migration (nothing here alters a table), so it isn't numbered
-- alongside 001-006. Run it yourself, once, in the Supabase SQL editor.
--
-- Scoped to jimgillon@gmail.com only, via an auth.users lookup by email —
-- never hardcode a uuid here. Guarded to be safe to re-run: every insert
-- first checks whether its row already exists (by owner + a natural key) and
-- skips it if so, rather than relying on a unique constraint that doesn't
-- exist on contacts or requests.
--
-- Dates are CURRENT_DATE-relative (INTERVAL offsets), not hardcoded, so the
-- Sent rows still land as a believable mix of open / upcoming / overdue /
-- done whenever this actually gets run, not only on 2026-08-08.
--
-- Received is deliberately NOT seeded — no schema/RLS path exists yet for a
-- signed-in recipient to see a Request sent to them (flagged separately,
-- unresolved). The Main Screen's Received subcard shows its own coming-soon
-- state regardless of what's in this table.
--
-- One live Dialog entry is seeded (on the Jack Smith request) so the Sent
-- row's "Dialog present" info icon has something real to reflect — Dialog is
-- fully built and RLS-enforced, unlike Attachments (still deferred per
-- CLAUDE.md Scope discipline), so no attachment-present demo data is seeded;
-- that icon simply won't appear on any row yet.
-- ============================================================================

do $$
declare
  v_owner     uuid;
  v_display   text;
  v_jack      uuid;
  v_jill      uuid;
  v_beatrice  uuid;
  v_rita      uuid;
  v_cat_wyp   uuid;
  v_cat_dev   uuid;
  v_cat_fin   uuid;
  v_req_jack  uuid;
begin
  select id into v_owner from auth.users where email = 'jimgillon@gmail.com';
  if v_owner is null then
    raise exception 'No auth.users row for jimgillon@gmail.com — sign in once first, then re-run.';
  end if;

  select display_name into v_display from public.profiles where id = v_owner;
  if v_display is null or trim(v_display) = '' then
    v_display := 'Jim';
  end if;

  -- ------------------------------------------------------------------------
  -- Contacts (Sent recipients)
  -- ------------------------------------------------------------------------
  select id into v_jack from public.contacts
    where owner_id = v_owner and email = 'jack.smith.demo@example.com';
  if v_jack is null then
    insert into public.contacts (owner_id, display_name, email, send_by)
    values (v_owner, 'Jack Smith', 'jack.smith.demo@example.com', 'email')
    returning id into v_jack;
  end if;

  select id into v_jill from public.contacts
    where owner_id = v_owner and email = 'jill.smith.demo@example.com';
  if v_jill is null then
    insert into public.contacts (owner_id, display_name, email, send_by)
    values (v_owner, 'Jill Smith', 'jill.smith.demo@example.com', 'email')
    returning id into v_jill;
  end if;

  select id into v_beatrice from public.contacts
    where owner_id = v_owner and email = 'beatrice.barrington.demo@example.com';
  if v_beatrice is null then
    insert into public.contacts (owner_id, display_name, email, send_by)
    values (v_owner, 'Beatrice Barrington', 'beatrice.barrington.demo@example.com', 'email')
    returning id into v_beatrice;
  end if;

  select id into v_rita from public.contacts
    where owner_id = v_owner and email = 'rita.anderson.demo@example.com';
  if v_rita is null then
    insert into public.contacts (owner_id, display_name, email, send_by)
    values (v_owner, 'Rita Anderson', 'rita.anderson.demo@example.com', 'email')
    returning id into v_rita;
  end if;

  -- ------------------------------------------------------------------------
  -- Categories (ToDos) — real unique index exists (categories_owner_name_ci_idx),
  -- so ON CONFLICT is safe here, unlike contacts/requests above and below.
  -- ------------------------------------------------------------------------
  insert into public.categories (owner_id, name)
    values (v_owner, 'Would You Please')
    on conflict (owner_id, (lower(name))) do nothing;
  select id into v_cat_wyp from public.categories
    where owner_id = v_owner and lower(name) = lower('Would You Please');

  insert into public.categories (owner_id, name)
    values (v_owner, 'Future Dev')
    on conflict (owner_id, (lower(name))) do nothing;
  select id into v_cat_dev from public.categories
    where owner_id = v_owner and lower(name) = lower('Future Dev');

  insert into public.categories (owner_id, name)
    values (v_owner, 'Personal Fin')
    on conflict (owner_id, (lower(name))) do nothing;
  select id into v_cat_fin from public.categories
    where owner_id = v_owner and lower(name) = lower('Personal Fin');

  -- ------------------------------------------------------------------------
  -- Sent Requests (contact_id not null). due_date/created_at are relative to
  -- today so Open/Overdue/Done stay meaningful on whatever day this runs.
  -- ------------------------------------------------------------------------

  -- 1. Open, due soon. Also gets a seeded Question so its "Dialog present"
  --    icon is live rather than staged.
  if not exists (
    select 1 from public.requests
    where owner_id = v_owner and description = 'Prepare the June cash flow report for Ally Chemical'
  ) then
    insert into public.requests (owner_id, contact_id, description, due_date, created_at)
    values (
      v_owner, v_jack, 'Prepare the June cash flow report for Ally Chemical',
      current_date + 5, now() - interval '21 days'
    )
    returning id into v_req_jack;

    insert into public.dialog (request_id, kind, body, who, author_user_id)
    values (
      v_req_jack, 'question',
      'Do you need this broken out by division, or just the company total?',
      v_display, v_owner
    );
  end if;

  -- 2. Overdue — due date in the past, never done.
  if not exists (
    select 1 from public.requests
    where owner_id = v_owner and description = 'Hire referees for Valley Little League'
  ) then
    insert into public.requests (owner_id, contact_id, description, due_date, created_at)
    values (
      v_owner, v_jill, 'Hire referees for Valley Little League',
      current_date - 21, now() - interval '2 months'
    );
  end if;

  -- 3. Open, due soon.
  if not exists (
    select 1 from public.requests
    where owner_id = v_owner and description = 'Send Lease Termination letter unless all back rent is paid.'
  ) then
    insert into public.requests (owner_id, contact_id, description, due_date, created_at)
    values (
      v_owner, v_beatrice, 'Send Lease Termination letter unless all back rent is paid.',
      current_date + 2, now() - interval '1 month'
    );
  end if;

  -- 4. Done, on time — exercises the Done-row styling and the Done column.
  if not exists (
    select 1 from public.requests
    where owner_id = v_owner and description = 'Confirm the office lease renewal terms with the landlord'
  ) then
    insert into public.requests (owner_id, contact_id, description, due_date, done_date, created_at)
    values (
      v_owner, v_rita, 'Confirm the office lease renewal terms with the landlord',
      current_date - 21, current_date - 21, now() - interval '2 months'
    );
  end if;

  -- ------------------------------------------------------------------------
  -- ToDos (contact_id null). Priority is required for a ToDo; due_date is
  -- optional and the Main Screen ToDos row doesn't render it, but a couple
  -- are populated anyway for realism.
  -- ------------------------------------------------------------------------

  if not exists (
    select 1 from public.requests
    where owner_id = v_owner and description = 'Finish the PRD for the WYP Request and ToDo Application'
      and contact_id is null
  ) then
    insert into public.requests (owner_id, category_id, description, priority, due_date)
    values (
      v_owner, v_cat_wyp, 'Finish the PRD for the WYP Request and ToDo Application', 1,
      current_date + 3
    );
  end if;

  if not exists (
    select 1 from public.requests
    where owner_id = v_owner
      and description = 'Respond to dev quote request for the new user interface related to mailing list preparation and subscriber onboarding flows'
      and contact_id is null
  ) then
    insert into public.requests (owner_id, category_id, description, priority, due_date)
    values (
      v_owner, v_cat_dev,
      'Respond to dev quote request for the new user interface related to mailing list preparation and subscriber onboarding flows',
      1, current_date + 1
    );
  end if;

  if not exists (
    select 1 from public.requests
    where owner_id = v_owner and description = 'Prepare a budget for an August fishing trip to Alaska'
      and contact_id is null
  ) then
    insert into public.requests (owner_id, category_id, description, priority, due_date)
    values (v_owner, v_cat_fin, 'Prepare a budget for an August fishing trip to Alaska', 2, current_date + 14);
  end if;

  if not exists (
    select 1 from public.requests
    where owner_id = v_owner and description = 'Choose a C Corp filing preparation company for Would You Please'
      and contact_id is null
  ) then
    insert into public.requests (owner_id, category_id, description, priority)
    values (v_owner, v_cat_wyp, 'Choose a C Corp filing preparation company for Would You Please', 3);
  end if;

end $$;
